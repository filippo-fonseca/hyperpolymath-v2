```typescript
import { requireOnboarded } from "@/lib/auth/get-user";
import { format } from "date-fns";
import { listFoodLogsForDay, getFoodHistory, getNutritionTargets } from "@/lib/nutrition/nutrition-service";
import { NutritionClient } from "@/components/nutrition/NutritionClient";

export default async function NutritionPage() {
  const user = await requireOnboarded();
  const today = format(new Date(), "yyyy-MM-dd"); // local TZ — Pitfall 6
  const [initialLogs, foodHistory, targets] = await Promise.all([
    listFoodLogsForDay(user.id, today),
    getFoodHistory(user.id, { limit: 20 }),
    getNutritionTargets(user.id),
  ]);
  return (
    <NutritionClient
      userId={user.id}
      initialDate={today}
      initialLogs={initialLogs}
      foodHistory={foodHistory}
      targets={targets}
    />
  );
}
```
