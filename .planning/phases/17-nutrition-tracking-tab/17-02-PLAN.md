```typescript
export type MacroNumbers = { kcal: number; proteinG: number; carbsG: number; fatG: number; fiberG?: number | null };
export type FoodMacroSource = { kcalPer100g: string | number; proteinPer100g: string | number; carbsPer100g: string | number; fatPer100g: string | number; fiberPer100g?: string | number | null };
export type NutritionTargetPcts = { targetKcal: number; proteinPct: number; carbsPct: number; fatPct: number };

const n = (v: string | number) => typeof v === "number" ? v : parseFloat(v);
const round1 = (x: number) => Math.round(x * 10) / 10;

export function resolveBaseAmount(quantity: number, servingGramsOrMl: number): number {
  return quantity * servingGramsOrMl;
}

export function computeMacros(baseAmount: number, food: FoodMacroSource): MacroNumbers {
  const factor = baseAmount / 100;
  return {
    kcal: Math.round(n(food.kcalPer100g) * factor),
    proteinG: round1(n(food.proteinPer100g) * factor),
    carbsG: round1(n(food.carbsPer100g) * factor),
    fatG: round1(n(food.fatPer100g) * factor),
    fiberG: food.fiberPer100g != null ? round1(n(food.fiberPer100g) * factor) : null,
  };
}

export function validateMacroConsistency(m: MacroNumbers): boolean {
  const derived = m.proteinG * 4 + m.carbsG * 4 + m.fatG * 9;
  if (m.kcal <= 0) return derived <= 5;
  return Math.abs(derived - m.kcal) / m.kcal < 0.15; // ±15% tolerance per RESEARCH
}

export function deriveTargetGrams(t: NutritionTargetPcts) {
  return {
    proteinG: Math.round((t.targetKcal * t.proteinPct / 100) / 4),
    carbsG: Math.round((t.targetKcal * t.carbsPct / 100) / 4),
    fatG: Math.round((t.targetKcal * t.fatPct / 100) / 9),
  };
}
```

```typescript
import { z } from "zod";

const OFF_USER_AGENT = "hyperpolymath-v2/1.0 (filifonsecacagnazzo@gmail.com)";
const SEARCH_URL = "https://search.openfoodfacts.org/search";
const PRODUCT_URL = "https://world.openfoodfacts.org/api/v2/product";

const NutrimentsSchema = z.object({
  "energy-kcal_100g": z.number().optional().default(0),
  "proteins_100g": z.number().optional().default(0),
  "carbohydrates_100g": z.number().optional().default(0),
  "fat_100g": z.number().optional().default(0),
  "fiber_100g": z.number().optional(),
  "sodium_100g": z.number().optional(),
}).passthrough();

const OffProductSchema = z.object({
  code: z.string().optional(),
  product_name: z.string().optional().default(""),
  brands: z.string().optional(),
  nutriments: NutrimentsSchema.optional().default({}),
  serving_size: z.string().optional(),
  serving_quantity: z.union([z.number(), z.string()]).optional(),
  product_quantity_unit: z.string().optional(),
  quantity: z.string().optional(),
});

export type OffProduct = z.infer<typeof OffProductSchema>;

export function normalizeOffProduct(raw: unknown) {
  const parsed = OffProductSchema.parse(raw);
  const n = parsed.nutriments;
  const baseUnit: "g" | "ml" = parsed.product_quantity_unit === "ml" ? "ml" : "g";
  const servingQuantity = typeof parsed.serving_quantity === "string"
    ? parseFloat(parsed.serving_quantity) || null
    : parsed.serving_quantity ?? null;
  return {
    offBarcode: parsed.code ?? null,
    name: parsed.product_name || "Unknown product",
    brand: parsed.brands?.split(",")[0]?.trim() || null,
    kcalPer100g: n["energy-kcal_100g"] ?? 0,
    proteinPer100g: n["proteins_100g"] ?? 0,
    carbsPer100g: n["carbohydrates_100g"] ?? 0,
    fatPer100g: n["fat_100g"] ?? 0,
    fiberPer100g: n["fiber_100g"] ?? null,
    sodiumPer100g: n["sodium_100g"] ?? null,
    baseUnit,
    servingSizeLabel: parsed.serving_size ?? null,
    servingQuantity, // grams or ml for "1 serving"
  };
}

export async function offSearch(q: string, opts: { revalidate?: number } = {}) {
  const url = `${SEARCH_URL}?q=${encodeURIComponent(q)}&json=1&page_size=20&fields=code,product_name,brands,nutriments,serving_size,serving_quantity,product_quantity_unit`;
  const res = await fetch(url, {
    headers: { "User-Agent": OFF_USER_AGENT },
    next: { revalidate: opts.revalidate ?? 300 },
  });
  if (!res.ok) throw new Error(`OFF search failed: ${res.status}`);
  const data = await res.json();
  return { hits: (data.hits ?? []).map(normalizeOffProduct), count: data.count ?? 0 };
}

export async function offProduct(barcode: string, opts: { revalidate?: number } = {}) {
  const url = `${PRODUCT_URL}/${encodeURIComponent(barcode)}.json?fields=code,product_name,brands,nutriments,serving_size,serving_quantity,product_quantity_unit,quantity`;
  const res = await fetch(url, {
    headers: { "User-Agent": OFF_USER_AGENT },
    next: { revalidate: opts.revalidate ?? 3600 },
  });
  if (!res.ok) throw new Error(`OFF product failed: ${res.status}`);
  const data = await res.json();
  if (data.status !== 1) return null;
  return normalizeOffProduct(data.product);
}
```

Required exports:

- `deleteLog(userId, logId)` — DELETE FROM food_logs WHERE id = logId AND user_id = userId
- `updateLog(userId, logId, patch)` — patch: `{ quantity?, servingOptionId?, mealSlot?, notes? }`. Re-snapshot macros if quantity/serving changes. Double-WHERE.
- `logMeal(userId, mealId, date, mealSlot)` — read meal + meal_items + foods (all double-WHERE), produce N food_logs rows in one transaction, bump meals.useCount + lastUsedAt.
- `upsertFood(userId, input)` — input: normalized OFF product OR manual input. If offBarcode present: ON CONFLICT (user_id, off_barcode) DO UPDATE (refresh macros). Always seed `food_serving_options` "100 g" or "100 ml" default; if input.servingQuantity > 0, seed product serving as additional option with `isDefault: true` (and demote 100g to non-default). Returns `{ food, servingOptions }`.
- `createMeal(userId, input)` — input: `{ name, description?, items: Array<{ foodId, servingOptionId, quantity, orderIndex }> }`. Insert meal + meal_items in transaction. All foodIds + servingOptionIds verified against userId.
- `listFoodLogsForDay(userId, date)` — SELECT food_logs JOIN foods + serving_options WHERE user_id = userId AND log_date = date ORDER BY meal_slot, created_at.
- `getFoodHistory(userId, { limit?, query? })` — SELECT foods WHERE user_id = userId, optional ilike on name, ORDER BY last_used_at DESC NULLS LAST, use_count DESC LIMIT (default 20). For D-02 "history-first" search.

```typescript
import { requireOnboarded } from "@/lib/auth/get-user";
import { offSearch } from "@/lib/nutrition/off-client";

export const runtime = "nodejs";

export async function GET(req: Request) {
  await requireOnboarded(); // auth gate
  const q = new URL(req.url).searchParams.get("q");
  if (!q || q.length < 2) return NextResponse.json({ hits: [] });
  try {
    const result = await offSearch(q);
    return NextResponse.json(result);
  } catch (err) {
    return NextResponse.json({ hits: [], error: String(err) }, { status: 502 });
  }
}
```
