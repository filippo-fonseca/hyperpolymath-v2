```sql
ALTER TABLE foods ENABLE ROW LEVEL SECURITY;

CREATE POLICY foods_owner_update ON foods FOR UPDATE USING (user_id = auth.uid()) WITH CHECK (user_id = auth.uid());
CREATE POLICY foods_owner_delete ON foods FOR DELETE USING (user_id = auth.uid());
```
