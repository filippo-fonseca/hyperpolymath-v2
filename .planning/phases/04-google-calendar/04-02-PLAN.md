    **Step 3 — `apps/web/lib/db/queries/gcal-connection.ts`** (read-only helper for Settings + future UI):
    ```typescript
    import { db } from "@/lib/db";
    import { users } from "@/lib/db/schema";
    import { eq } from "drizzle-orm";

    export type GcalConnectionStatus = "connected" | "not_connected" | "expired";

    export async function getGcalConnectionStatus(userId: string): Promise<GcalConnectionStatus> {
      const rows = await db
        .select({
          refreshEnc: users.gcalRefreshTokenEncrypted,
          expiresAt: users.gcalTokenExpiresAt,
        })
        .from(users)
        .where(eq(users.id, userId))
        .limit(1);
      if (!rows[0]?.refreshEnc) return "not_connected";
      // "Expired" here means access token has lapsed by >5 minutes AND we haven't refreshed yet.
      // Since getValidGcalToken refreshes lazily, "expired" status is mostly observable post-revoke
      // (when getValidGcalToken clears the columns and sets status back to "not_connected").
      // A genuine "expired but refreshable" state is rare in practice; show "connected" if a refresh
      // token exists. The /calendar banner (Plan 04-03/04) handles GcalTokenRevokedError separately.
      return "connected";
    }
    ```
    Status keeps it simple: refresh token exists → connected; missing → not connected. The `"expired"` literal is reserved for a future surfacing if needed (e.g., if a background tick proactively detects expiry without a user-triggered refresh).

    **Step 4 — `apps/web/app/actions/gcal-connection.ts`** (Server Action — disconnect):
    ```typescript
    "use server";
    import { google } from "googleapis";
    import { eq } from "drizzle-orm";
    import { db } from "@/lib/db";
    import { users } from "@/lib/db/schema";
    import { getUserOrRedirect } from "@/lib/auth/get-user";
    import { decryptToken } from "@/lib/gcal/encryption";
    import { createOAuth2Client } from "@/lib/gcal/client";

    type ActionResult = { success: true } | { success: false; error: string };

    /**
     * CAL-09: Disconnect Google Calendar.
     * Per RESEARCH Pitfall 6: clear DB columns FIRST, then best-effort revokeToken.
     * The opposite order (revoke first) risks "Google revoked but DB shows connected".
     * This order risks "DB cleared but Google still shows the token as valid for a while"
     * which is the safer failure mode (user re-consents on next Connect, Google issues fresh tokens
     * via prompt=consent, and the stale token will lapse on its own).
     */
    export async function disconnectGcal(): Promise<ActionResult> {
      const user = await getUserOrRedirect();

      // Step A: read current refresh token (for the best-effort revoke).
      const rows = await db
        .select({ refreshEnc: users.gcalRefreshTokenEncrypted })
        .from(users)
        .where(eq(users.id, user.id))
        .limit(1);
      const refreshToken = rows[0]?.refreshEnc ? decryptToken(rows[0].refreshEnc) : null;

      // Step B: CLEAR DB FIRST (Pitfall 6).
      await db.update(users).set({
        gcalRefreshTokenEncrypted: null,
        gcalAccessTokenEncrypted: null,
        gcalTokenExpiresAt: null,
        // D-09 / D-10: leave gcal_default_calendar_id + gcal_visible_calendar_ids in place —
        // a reconnect with the same Google account is the common case; preserving these
        // settings avoids re-prompting the user. They are cleared explicitly only if the
        // reconnect target a different Google account (handled in Plan 04-04 callback comparison).
      }).where(eq(users.id, user.id));

      // Step C: best-effort revoke (do NOT throw on failure).
      if (refreshToken) {
        try {
          const oauth2Client = createOAuth2Client();
          await oauth2Client.revokeToken(refreshToken);
        } catch (e) {
          // Log but don't surface — the DB is already cleared; user is disconnected locally.
          console.warn("[gcal] revokeToken failed (best-effort):", e);
        }
      }
      return { success: true };
    }
    ```

    "use client";
    import { useState, useTransition, useEffect } from "react";
    import { useSearchParams, useRouter } from "next/navigation";
    import { toast } from "sonner";
    import { Button } from "@/components/ui/button";
    import {
      AlertDialog, AlertDialogTrigger, AlertDialogContent, AlertDialogHeader,
      AlertDialogTitle, AlertDialogDescription, AlertDialogFooter,
      AlertDialogAction, AlertDialogCancel,
    } from "@/components/ui/alert-dialog";
    import { disconnectGcal } from "@/app/actions/gcal-connection";
    import type { GcalConnectionStatus } from "@/lib/db/queries/gcal-connection";

    interface Props { status: GcalConnectionStatus; }

    const STATUS_COPY = {
      connected:     { label: "Connected",     dot: "bg-green-500" },
      not_connected: { label: "Not connected", dot: "bg-neutral-400" },
      expired:       { label: "Token expired", dot: "bg-amber-500" },
    } as const;

    export function GcalConnectionRow({ status }: Props) {
      const [pending, startTransition] = useTransition();
      const router = useRouter();
      const params = useSearchParams();

      // Surface ?gcal= query params from the OAuth callback as toasts. Then strip them
      // (router.replace) so a refresh doesn't re-fire.
      useEffect(() => {
        const flag = params.get("gcal");
        if (!flag) return;
        if (flag === "denied") toast.error("Google Calendar connection cancelled.");
        else if (flag === "invalid_state") toast.error("Connection failed: state mismatch. Try again.");
        else if (flag === "no_refresh_token") toast.error("Google didn't return a refresh token. Disconnect & retry.");
        // NOTE (m-03 fix): no `connected` branch here. The OAuth callback redirects to
        // `/calendar?gcal=connected` on success — never to /settings — so the "connected"
        // toast lives in CalendarClient.tsx (Plan 04-03 Task 2 Step 4), not here.
        // Only error/cancel paths land back at /settings.
        router.replace("/settings");
      }, [params, router]);

      const handleDisconnect = () => {
        startTransition(async () => {
          const res = await disconnectGcal();
          if (res.success) toast.success("Google Calendar disconnected.");
          else toast.error(res.error ?? "Failed to disconnect.");
          router.refresh(); // Force Server Component re-fetch of status (no Realtime on /settings — manual refresh OK)
        });
      };

      const meta = STATUS_COPY[status];

      return (
        <div className="flex items-center justify-between gap-4 py-4 border-b border-border">
          <div className="flex flex-col gap-1">
            <div className="text-sm font-medium">Google Calendar</div>
