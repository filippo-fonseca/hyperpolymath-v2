<optimistic_reducer_decision>
**Decision: REUSE Phase 3's `optimisticReducer<T extends { id: string }>` — do NOT create a separate `optimistic-events-reducer.ts`.**

Reasoning (per RESEARCH Pitfall 7):
- The Phase 3 reducer is already generic over any `T extends { id: string }`. It does not assume UUID format.
- The Phase 4 wiring differs from Phase 3, but the reducer algebra is unchanged.
- Phase 3 wiring (UUID dedupe on echo): `insert(uuid) → server persists with same uuid → echo carries same id → reducer dedupes on id match`.
- Phase 4 wiring (canonical ID swap on success): `insert(placeholderId) → server returns canonical gcal id → dispatch delete(placeholderId) + insert(canonicalEvent) → reducer handles both ops`.
- Single source of truth for the reducer. The differing usage patterns are documented at the call sites.

The dedicated test file `gcal-events-reducer.test.ts` covers the gcal-shaped-ID-handling explicitly (RESEARCH §Code Examples — last test in that file: "dedupes by string ID regardless of UUID format").
</optimistic_reducer_decision>

<drag_resize_behavior_decision>
**Decision: Auto-save on drag-end (no Sheet on resize).**

Per CONTEXT.md D-01 ambiguity ("drag-resize on existing event edges extends/shortens duration"), and per planner_directive item 6: drag-resize auto-saves on resize-end with optimistic update. The Sheet does NOT open on resize — that would be jarring. Drag-move (whole-event drag) also auto-saves on drop-end. The Sheet remains the canonical surface for create + click-edit + delete.

Failure mode: optimistic update applies → Server Action `updateEvent` calls `events.patch` → on success, refetch confirms; on failure, revert + `toast.error("Failed to reschedule — change reverted")`.
</drag_resize_behavior_decision>

<events_move_decision>
**Decision: Sheet save handles changing `calendarId` via `events.move`; drag-move keeps the source calendarId (defer cross-calendar drag).**

Per RESEARCH Open Q 3: drag-move stays within the source calendar. If the user wants to change the calendar, they open the Sheet and pick a different option from the dropdown. The save action detects `form.calendarId !== state.event.calendarId` and routes to `events.move` instead of `events.patch`. The grid's `onEventDrop` only ever changes start/end, never calendarId.
</events_move_decision>

    const CreateEventSchema = z.object({
      calendarId: z.string().min(1),
      title: z.string().min(1).max(1024),
      description: z.string().max(8192).optional().nullable(),
      // ISO with offset; allDay flag determines whether to use date vs dateTime in request
      start: z.string().min(1),
      end: z.string().min(1),
      allDay: z.boolean(),
      userTimezone: z.string().min(1), // e.g., "America/New_York" — used for non-all-day start/end
    });

    export async function createEvent(input: unknown): Promise<ActionResult<GcalEventDTO>> {
      const user = await requireOnboarded();
      const parsed = CreateEventSchema.safeParse(input);
      if (!parsed.success) return { success: false, error: parsed.error.issues[0]?.message ?? "Invalid input", kind: "unknown" };

      let cal;
      try { cal = await getValidGcalToken(user.id); }
      catch (e) {
        if (e instanceof GcalTokenRevokedError) return { success: false, error: "Reconnect Google Calendar", kind: "revoked" };
        if (e instanceof GcalNotConnectedError) return { success: false, error: "Not connected", kind: "not_connected" };
        throw e;
      }

      const { calendarId, title, description, start, end, allDay, userTimezone } = parsed.data;

      const requestBody: calendar_v3.Schema$Event = {
        summary: title,
        description: description ?? undefined,
        start: allDay ? { date: start.slice(0, 10) } : { dateTime: start, timeZone: userTimezone },
        end: allDay ? { date: end.slice(0, 10) } : { dateTime: end, timeZone: userTimezone },
      };

      const { data } = await cal.events.insert({ calendarId, requestBody });
      const dto = eventToDTO(data, calendarId);
      if (!dto) return { success: false, error: "gcal returned invalid event shape", kind: "unknown" };
      return { success: true, data: dto };
    }

    // Update — handles both events.patch (same calendar) and events.move (calendar changed)
    const UpdateEventSchema = z.object({
      eventId: z.string().min(1),
      currentCalendarId: z.string().min(1),
      newCalendarId: z.string().min(1), // may equal currentCalendarId
      title: z.string().min(1).max(1024).optional(),
      description: z.string().max(8192).nullable().optional(),
      start: z.string().min(1).optional(),
      end: z.string().min(1).optional(),
      allDay: z.boolean().optional(),
      userTimezone: z.string().min(1),
    });

    export async function updateEvent(input: unknown): Promise<ActionResult<GcalEventDTO>> {
      const user = await requireOnboarded();
      const parsed = UpdateEventSchema.safeParse(input);
      if (!parsed.success) return { success: false, error: parsed.error.issues[0]?.message ?? "Invalid input", kind: "unknown" };

      let cal;
      try { cal = await getValidGcalToken(user.id); }
      catch (e) {
        if (e instanceof GcalTokenRevokedError) return { success: false, error: "Reconnect Google Calendar", kind: "revoked" };
        if (e instanceof GcalNotConnectedError) return { success: false, error: "Not connected", kind: "not_connected" };
        throw e;
      }

      const { eventId, currentCalendarId, newCalendarId, title, description, start, end, allDay, userTimezone } = parsed.data;

      // If calendar changed, move first (events.move retains the event ID per gcal docs)
      let effectiveCalendarId = currentCalendarId;
      if (newCalendarId !== currentCalendarId) {
        await cal.events.move({ calendarId: currentCalendarId, eventId, destination: newCalendarId });
        effectiveCalendarId = newCalendarId;
      }

      // Then patch field-level changes. Build requestBody only with defined fields.
      const requestBody: Partial<calendar_v3.Schema$Event> = {};
      if (title !== undefined) requestBody.summary = title;
      if (description !== undefined) requestBody.description = description ?? "";
      if (start !== undefined && end !== undefined && allDay !== undefined) {
        requestBody.start = allDay ? { date: start.slice(0, 10) } : { dateTime: start, timeZone: userTimezone };
        requestBody.end = allDay ? { date: end.slice(0, 10) } : { dateTime: end, timeZone: userTimezone };
      }

      const { data } = await cal.events.patch({ calendarId: effectiveCalendarId, eventId, requestBody });
      const dto = eventToDTO(data, effectiveCalendarId);
      if (!dto) return { success: false, error: "gcal returned invalid event shape", kind: "unknown" };
      return { success: true, data: dto };
    }

    const DeleteEventSchema = z.object({
      calendarId: z.string().min(1),
      eventId: z.string().min(1),
    });

    export async function deleteEvent(input: unknown): Promise<ActionResult<{ id: string }>> {
      const user = await requireOnboarded();
      const parsed = DeleteEventSchema.safeParse(input);
      if (!parsed.success) return { success: false, error: parsed.error.issues[0]?.message ?? "Invalid input", kind: "unknown" };

      let cal;
      try { cal = await getValidGcalToken(user.id); }
      catch (e) {
        if (e instanceof GcalTokenRevokedError) return { success: false, error: "Reconnect Google Calendar", kind: "revoked" };
        if (e instanceof GcalNotConnectedError) return { success: false, error: "Not connected", kind: "not_connected" };
        throw e;
      }

      await cal.events.delete({ calendarId: parsed.data.calendarId, eventId: parsed.data.eventId });
      return { success: true, data: { id: parsed.data.eventId } };
    }

        start: form.start,

      };
      startTransition(() => addOptimistic({ type: "insert", row: optimisticEvent }));

      const res = await createEvent({
        calendarId: form.calendarId,
        title: form.title,
        description: form.description ?? null,
        start: form.start.toISOString(),
        end: form.end.toISOString(),
        allDay: form.allDay,
        userTimezone: effectiveTz,
      });

      if (!res.success) {
        startTransition(() => addOptimistic({ type: "delete", id: placeholderId }));

    function swapPlaceholderForCanonical(placeholderId: string, dto: GcalEventDTO) {

      });

    }

    async function handleUpdate(eventId: string, currentCalendarId: string, patch: { newCalendarId?: string; title?: string; description?: string | null; start?: Date; end?: Date; allDay?: boolean; }) {
      // Optimistic update — apply the patch locally first.
      const patchForGrid: Partial<GcalEvent> = {};
      if (patch.title !== undefined) patchForGrid.title = patch.title;
      if (patch.description !== undefined) patchForGrid.description = patch.description;
      if (patch.start) patchForGrid.start = patch.start;
      if (patch.end) patchForGrid.end = patch.end;
      if (patch.allDay !== undefined) patchForGrid.allDay = patch.allDay;
      if (patch.newCalendarId) patchForGrid.calendarId = patch.newCalendarId;

      const previous = optimisticEvents.find((e) => e.id === eventId);
      startTransition(() => addOptimistic({ type: "update", id: eventId, patch: patchForGrid }));

      const res = await updateEvent({
        eventId,
        currentCalendarId,
        newCalendarId: patch.newCalendarId ?? currentCalendarId,
        title: patch.title,
        description: patch.description,
        start: patch.start?.toISOString(),
        end: patch.end?.toISOString(),
        allDay: patch.allDay,
        userTimezone: effectiveTz,
      });

      if (!res.success) {
        // Revert
        if (previous) startTransition(() => addOptimistic({ type: "update", id: eventId, patch: previous }));
        toast.error(res.error ?? "Failed to update event");
        return;
      }
      void qc.invalidateQueries({ queryKey: ["calendar-events", userId] });
    }

    async function handleDelete(eventId: string, calendarId: string) {
      const previous = optimisticEvents.find((e) => e.id === eventId);
      startTransition(() => addOptimistic({ type: "delete", id: eventId }));
      const res = await deleteEvent({ eventId, calendarId });
      if (!res.success) {
        if (previous) startTransition(() => addOptimistic({ type: "insert", row: previous }));
        toast.error(res.error ?? "Failed to delete event");
        return;
      }
      void qc.invalidateQueries({ queryKey: ["calendar-events", userId] });
    }

      calendars={calendars}
      defaultCalendarId={/* fetch from server props — Step 4 below */}
