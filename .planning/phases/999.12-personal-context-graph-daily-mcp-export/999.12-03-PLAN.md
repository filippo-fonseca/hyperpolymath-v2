    3. Create `packages/personal-context-mcp/src/tools/get-current-context.ts`:

       ```ts
       import { z } from 'zod';
       import type { McpServer } from '@modelcontextprotocol/sdk/server/mcp.js';
       import type { LoadSnapshot } from '../data/load-snapshot.js';
       import type { ContextSnapshot } from '../types.js';

       const TopicEnum = z.enum(['area','project','task','capture','training_activity','habit','jarvis_fact']);

       export function registerGetCurrentContext(
         server: McpServer,
         ctx: { userId: string; loadSnapshot: LoadSnapshot },
       ) {
         server.tool(
           'get_current_context',
           'Returns the most recent personal context snapshot for the connected user. ' +
           'Optionally filter to specific node types (topics). Use this at the start of ' +
           'any conversation to load what the agent already knows about the user.',
           { topics: z.array(TopicEnum).optional().describe('If provided, return only nodes of these types.') },
           async ({ topics }) => {
             const snap = await ctx.loadSnapshot(ctx.userId);
             if (!snap) return { content: [{ type: 'text', text: 'No snapshot available yet.' }] };
             const filtered: ContextSnapshot = topics?.length
               ? { ...snap, nodes: snap.nodes.filter(n => topics.includes(n.type)) }
               : snap;
             return { content: [{ type: 'text', text: JSON.stringify(filtered, null, 2) }] };
           },
         );
       }
       ```

    4. Create `packages/personal-context-mcp/src/tools/get-snapshot-history.ts`:

       ```ts
       import { z } from 'zod';
       import type { McpServer } from '@modelcontextprotocol/sdk/server/mcp.js';
       import type { LoadHistory } from '../data/load-snapshot.js';

       export function registerGetSnapshotHistory(
         server: McpServer,
         ctx: { userId: string; loadHistory: LoadHistory },
       ) {
         server.tool(
           'get_snapshot_history',
           'Returns metadata (date + per-type node counts) for recent snapshots. ' +
           'Does NOT include the full payload (those can be large) — call get_current_context for the latest payload.',
           { days: z.number().int().min(1).max(30).default(7).describe('Number of days of history (1..30).') },
           async ({ days }) => {
             const history = await ctx.loadHistory(ctx.userId, days);
             return { content: [{ type: 'text', text: JSON.stringify(history, null, 2) }] };
           },
         );
       }
       ```

    5. Create `packages/personal-context-mcp/src/index.ts`:

       ```ts
       import { McpServer } from '@modelcontextprotocol/sdk/server/mcp.js';
       import { registerGetCurrentContext } from './tools/get-current-context.js';
       import { registerGetSnapshotHistory } from './tools/get-snapshot-history.js';
       import type { LoadSnapshot, LoadHistory } from './data/load-snapshot.js';

       export type CreateServerOptions = {
         userId: string;
         loadSnapshot: LoadSnapshot;
         loadHistory: LoadHistory;
         serverName?: string;
         serverVersion?: string;
       };

       export function createPersonalContextServer(opts: CreateServerOptions): McpServer {
         const server = new McpServer({
           name: opts.serverName ?? 'hyperpolymath-personal-context',
           version: opts.serverVersion ?? '0.1.0',
         });
         registerGetCurrentContext(server, { userId: opts.userId, loadSnapshot: opts.loadSnapshot });
         registerGetSnapshotHistory(server, { userId: opts.userId, loadHistory: opts.loadHistory });
         return server;
       }

       export type { LoadSnapshot, LoadHistory } from './data/load-snapshot.js';
       export * from './types.js';
       ```
