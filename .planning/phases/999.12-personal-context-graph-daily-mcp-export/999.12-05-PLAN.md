       ```tsx
       import { desc, eq } from 'drizzle-orm';
       import { db } from '@/lib/db';
       import { personalContextSnapshots } from '@/lib/db/schema';
       import { createClient } from '@/lib/supabase/server'; // mirror existing import path
       import { ContextSnapshotClient } from './ContextSnapshotClient';

       export const dynamic = 'force-dynamic';

       export default async function ContextSettingsPage() {
         const supabase = await createClient();
         const { data: claims } = await supabase.auth.getClaims();
         const userId = claims?.claims?.sub as string | undefined;
         if (!userId) return null; // layout guard handles redirect

         const recent = await db.select({
             snapshotDate: personalContextSnapshots.snapshotDate,
             schemaVersion: personalContextSnapshots.schemaVersion,
             payload: personalContextSnapshots.payload,
             createdAt: personalContextSnapshots.createdAt,
           })
           .from(personalContextSnapshots)
           .where(eq(personalContextSnapshots.userId, userId))
           .orderBy(desc(personalContextSnapshots.snapshotDate))
           .limit(30);

         const latest = recent[0] ?? null;
         const historyMeta = recent.map((r) => {
           const p = r.payload as { meta?: { totalNodes?: number; totalEdges?: number; nodeCounts?: Record<string, number> } };
           return {
             snapshotDate: String(r.snapshotDate),
             totalNodes: p.meta?.totalNodes ?? 0,
             totalEdges: p.meta?.totalEdges ?? 0,
             nodeCounts: p.meta?.nodeCounts ?? {},
           };
         });

         return <ContextSnapshotClient latest={latest} historyMeta={historyMeta} />;
       }
       ```

    2. Create `apps/web/app/(app)/settings/mcp-tokens/actions.ts`:

       ```ts
       'use server';
       import { createClient } from '@/lib/supabase/server';
       import { mintMcpToken, revokeMcpToken } from '@/lib/mcp/mint-token';

       export async function mintMcpTokenAction(name: string) {
         const supabase = await createClient();
         const { data: claims } = await supabase.auth.getClaims();
         const userId = claims?.claims?.sub as string | undefined;
         if (!userId) return { ok: false as const, error: 'Unauthorized' };
         if (!name || name.length > 100) return { ok: false as const, error: 'Name required (max 100 chars).' };
         const r = await mintMcpToken(userId, name);
         if (!r.ok) return r;
         return { ok: true as const, plaintext: r.data.plaintext };
       }

       export async function revokeMcpTokenAction() {
         const supabase = await createClient();
         const { data: claims } = await supabase.auth.getClaims();
         const userId = claims?.claims?.sub as string | undefined;
         if (!userId) return { ok: false as const, error: 'Unauthorized' };
         return revokeMcpToken(userId);
       }
       ```

    1. Append a Server Action to `apps/web/app/(app)/settings/context/actions.ts` (or create the file if it doesn't exist yet from Task 1):

       ```ts
       'use server';
       import { and, eq } from 'drizzle-orm';
       import { db } from '@/lib/db';
       import { captures, tasks, jarvisFacts } from '@/lib/db/schema';
       import { createClient } from '@/lib/supabase/server';

       type Resource = 'capture' | 'task' | 'jarvis_fact';

       export async function setNoExportAction(resource: Resource, id: string, value: boolean) {
         const supabase = await createClient();
         const { data: claims } = await supabase.auth.getClaims();
         const userId = claims?.claims?.sub as string | undefined;
         if (!userId) return { ok: false as const, error: 'Unauthorized' };
         try {
           if (resource === 'capture') {
             await db.update(captures).set({ noExport: value })
               .where(and(eq(captures.id, id), eq(captures.userId, userId)));
           } else if (resource === 'task') {
             await db.update(tasks).set({ noExport: value })
               .where(and(eq(tasks.id, id), eq(tasks.userId, userId)));
           } else {
             await db.update(jarvisFacts).set({ noExport: value })
               .where(and(eq(jarvisFacts.id, id), eq(jarvisFacts.userId, userId)));
           }
           return { ok: true as const };
         } catch (e) {
           return { ok: false as const, error: e instanceof Error ? e.message : 'update failed' };
         }
       }
       ```

       (If the schema variable names differ — e.g. `jarvisFacts` vs `jarvis_facts` — match the actual exports in schema.ts. Read first.)

    2. Create `apps/web/components/privacy/NoExportToggle.tsx`:

       ```tsx
       'use client';
       import { useOptimistic, useTransition } from 'react';
       import { setNoExportAction } from '@/app/(app)/settings/context/actions';

       export type NoExportResource = 'capture' | 'task' | 'jarvis_fact';

       export function NoExportToggle({ resource, id, initial }: {
         resource: NoExportResource; id: string; initial: boolean;
       }) {
         const [optimistic, setOptimistic] = useOptimistic<boolean, boolean>(initial, (_s, next) => next);
         const [isPending, startTransition] = useTransition();
         return (
           <button
             type="button"
             aria-pressed={optimistic}
             disabled={isPending}
             onClick={() => {
               const next = !optimistic;
               startTransition(async () => {
                 setOptimistic(next);
                 const r = await setNoExportAction(resource, id, next);
                 if (!r.ok) setOptimistic(!next); // revert on failure
               });
             }}
             className="text-xs font-mono opacity-60 hover:opacity-100 transition-opacity"
             title={optimistic
               ? 'This item is hidden from the MCP export. Click to include.'
               : 'Hide this item from the MCP export.'}
           >
             {optimistic ? 'no-export · on' : 'no-export · off'}
           </button>
         );
       }
       ```

       Style is intentionally minimal/utilitarian — privacy controls should
       feel like a quiet toggle, not a marketing button. Refine to match the
