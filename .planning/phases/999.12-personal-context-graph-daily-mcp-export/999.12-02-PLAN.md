    import { type Result, ok, err } from '@/lib/integrations/result';
    import { ContextSnapshotSchema, CURRENT_SCHEMA_VERSION, type ContextSnapshot } from './types';

    export { CURRENT_SCHEMA_VERSION };

    export type LegacySnapshot = { _legacy: true; schemaVersion: number; [k: string]: unknown };

    export function migrate(payload: unknown, fromVersion: number): Result<ContextSnapshot | LegacySnapshot> {
      if (fromVersion === CURRENT_SCHEMA_VERSION) {
        const parsed = ContextSnapshotSchema.safeParse(payload);
        if (!parsed.success) return err(`v${CURRENT_SCHEMA_VERSION} payload failed validation: ${parsed.error.message}`);
        return ok(parsed.data);
      }
      if (fromVersion > CURRENT_SCHEMA_VERSION) {
        return err(`snapshot v${fromVersion} is newer than reader v${CURRENT_SCHEMA_VERSION}`);
      }
      // fromVersion < CURRENT: no migrators yet (we're at v1). Return opaque legacy.
      return ok({ _legacy: true, schemaVersion: fromVersion, ...(payload as object) } as LegacySnapshot);
    }
    ```

    // apps/web/lib/context/nodes/tasks.ts
    import { and, desc, eq, ne } from 'drizzle-orm';
    import { db as defaultDb } from '@/lib/db';
    import { tasks as tasksTable, tasksProjects } from '@/lib/db/schema';
    import type { Node } from '../types';

    export type DB = typeof defaultDb;

    export async function loadTasks(userId: string, db: DB = defaultDb): Promise<{ nodes: Node[]; excluded: number }> {
      // Most recent 200 non-lesno tasks where no_export=false.
      // Count of no_export=true rows for meta.
      const rows = await db.select({
          id: tasksTable.id, title: tasksTable.title, priority: tasksTable.priority,
          status: tasksTable.status, dueDate: tasksTable.dueDate, noExport: tasksTable.noExport,
        })
        .from(tasksTable)
        .where(and(eq(tasksTable.userId, userId), ne(tasksTable.status, 'lesno')))
        .orderBy(desc(tasksTable.updatedAt))
        .limit(250); // a bit over cap so excluded count is more accurate within the window

      // Junction is `tasksProjects` (table `tasks_projects`) with columns
      // `taskId`, `projectId`, `userId` — verified in schema.ts lines 217-230.
      const projectLinks = await db.select({ taskId: tasksProjects.taskId, projectId: tasksProjects.projectId })
        .from(tasksProjects).where(eq(tasksProjects.userId, userId));
      const byTask = new Map<string, string[]>();
      for (const l of projectLinks) {
        const arr = byTask.get(l.taskId) ?? [];
        arr.push(l.projectId);
        byTask.set(l.taskId, arr);
      }

      let excluded = 0;
      const nodes: Node[] = [];
      for (const r of rows) {
        if (r.noExport) { excluded++; continue; }
        if (nodes.length >= 200) continue;
        nodes.push({
          type: 'task', id: r.id, title: r.title,
          priority: r.priority as 'P∞'|'P1'|'P2'|'P3',
          status: r.status as 'not started'|'up next'|'in progress'|'almost done'|'lesno',
          dueDate: r.dueDate ? new Date(r.dueDate).toISOString() : null,
          projectIds: byTask.get(r.id) ?? [],
        });
      }
      return { nodes, excluded };
    }

    import type { Edge, Node } from './types';

    export function deriveEdges(input: {
      areaIds: Set<string>;
      projects: Extract<Node, { type: 'project' }>[];
      tasks: Extract<Node, { type: 'task' }>[];
      captures: Extract<Node, { type: 'capture' }>[];
      facts: { id: string; aboutEntityType?: 'area'|'project'; aboutEntityId?: string }[];
    }): Edge[] {
      const edges: Edge[] = [];
      for (const p of input.projects) {
        if (input.areaIds.has(p.areaId)) edges.push({ type: 'project_in_area', from: p.id, to: p.areaId });
      }
      for (const t of input.tasks) {
        for (const pid of t.projectIds) edges.push({ type: 'task_in_project', from: t.id, to: pid });
      }
      for (const c of input.captures) {
        for (const pid of c.projectIds) edges.push({ type: 'capture_in_project', from: c.id, to: pid });
        for (const tag of c.tags) edges.push({ type: 'capture_tagged', from: c.id, tag });
      }
      for (const f of input.facts) {
        if (f.aboutEntityType && f.aboutEntityId) {
          edges.push({ type: 'fact_about', from: f.id, entityType: f.aboutEntityType, entityId: f.aboutEntityId });
        }
      }
      return edges;
    }
    ```

    import { type Result, ok, err } from '@/lib/integrations/result';
    import { db as defaultDb } from '@/lib/db';
    import { ContextSnapshotSchema, CURRENT_SCHEMA_VERSION, type ContextSnapshot, type Node } from './types';
    import { loadAreas } from './nodes/areas';
    import { loadProjects } from './nodes/projects';
    import { loadTasks } from './nodes/tasks';
    import { loadCaptures } from './nodes/captures';
    import { loadTraining } from './nodes/training';
    import { loadHabits } from './nodes/habits';
    import { loadJarvisFacts } from './nodes/jarvis-facts';
    import { deriveEdges } from './edges';

    export { CURRENT_SCHEMA_VERSION };

    type DB = typeof defaultDb;

    export async function buildSnapshot(userId: string, db: DB = defaultDb): Promise<Result<ContextSnapshot>> {
      try {
        const [areas, projects, tasks, captures, training, habits, facts] = await Promise.all([
          loadAreas(userId, db),
          loadProjects(userId, db),
          loadTasks(userId, db),
          loadCaptures(userId, db),
          loadTraining(userId, db),
          loadHabits(userId, db),
          loadJarvisFacts(userId, db),
        ]);

        const allNodes: Node[] = [
          ...areas.nodes, ...projects.nodes, ...tasks.nodes,
          ...captures.nodes, ...training.nodes, ...habits.nodes, ...facts.nodes,
        ];

        const areaIds = new Set(areas.nodes.filter(n => n.type === 'area').map(n => n.id));
        const projectNodes = projects.nodes.filter((n): n is Extract<Node, { type: 'project' }> => n.type === 'project');
        const taskNodes = tasks.nodes.filter((n): n is Extract<Node, { type: 'task' }> => n.type === 'task');
        const captureNodes = captures.nodes.filter((n): n is Extract<Node, { type: 'capture' }> => n.type === 'capture');

        const edges = deriveEdges({
          areaIds, projects: projectNodes, tasks: taskNodes, captures: captureNodes,
          facts: facts.nodes.filter(n => n.type === 'jarvis_fact').map(n => ({ id: n.id })),
        });

        const nodeCounts: Record<string, number> = {};
        for (const n of allNodes) nodeCounts[n.type] = (nodeCounts[n.type] ?? 0) + 1;

        const snapshot: ContextSnapshot = {
          schemaVersion: CURRENT_SCHEMA_VERSION,
          generatedAt: new Date().toISOString(),
          nodes: allNodes,
          edges,
          meta: {
            totalNodes: allNodes.length,
            totalEdges: edges.length,
            nodeCounts,
            excludedNoExportCount: tasks.excluded + captures.excluded + facts.excluded,
          },
        };

        return ok(ContextSnapshotSchema.parse(snapshot));
      } catch (e) {
        return err(e instanceof Error ? e.message : 'snapshot build failed');
      }
    }
    ```

    import { db as defaultDb } from '@/lib/db';
    import { personalContextSnapshots } from '@/lib/db/schema';
    import { type Result, ok, err } from '@/lib/integrations/result';
    import { type ContextSnapshot, CURRENT_SCHEMA_VERSION } from './types';

    type DB = typeof defaultDb;

    /**
     * Upsert today's snapshot for `userId`. Idempotent on (user_id, snapshot_date) PK —
     * a same-day rebuild overwrites the prior row's payload but preserves created_at
     * (per CTX-01 PK semantics; created_at carries the first-write timestamp).
     *
     * `snapshotDate` defaults to today (UTC) so the cron and the manual "Rebuild now"
     * button hit the same row on the same calendar day.
     */
    export async function persistSnapshot(
      userId: string,
      snapshot: ContextSnapshot,
      opts: { db?: DB; snapshotDate?: string } = {},
    ): Promise<Result<{ userId: string; snapshotDate: string }>> {
      const db = opts.db ?? defaultDb;
      const snapshotDate = opts.snapshotDate ?? new Date().toISOString().slice(0, 10); // YYYY-MM-DD UTC
      try {
        await db.insert(personalContextSnapshots)
          .values({
            userId,
            snapshotDate,
            schemaVersion: CURRENT_SCHEMA_VERSION,
            payload: snapshot as unknown as Record<string, unknown>, // jsonb column typed loosely in schema.ts (see Plan 01 Task 3 note)
          })
          .onConflictDoUpdate({
            target: [personalContextSnapshots.userId, personalContextSnapshots.snapshotDate],
            set: {
              schemaVersion: CURRENT_SCHEMA_VERSION,
              payload: snapshot as unknown as Record<string, unknown>,
            },
          });
        return ok({ userId, snapshotDate });
      } catch (e) {
        return err(e instanceof Error ? e.message : 'persistSnapshot failed');
      }
    }
    ```
