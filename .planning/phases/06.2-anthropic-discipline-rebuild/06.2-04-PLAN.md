    import { cn } from '@/lib/utils';

    type ServiceStatus = 'ok' | 'down' | 'n/a';

    interface HealthService {
      name: string;
      status: ServiceStatus;
      latency?: number | null;
      note?: string;
    }

    interface HealthResponse {
      services: {
        supabase: { status: ServiceStatus; latency?: number | null };
        anthropic: { status: ServiceStatus; latency?: number | null };
        google_calendar: { status: ServiceStatus; note?: string };
      };
      checkedAt: string;
    }

    async function fetchHealth(): Promise<HealthResponse> {
      // Adapt to the actual fetch helper used in the codebase; do NOT change /api/health route
      const url = process.env.NEXT_PUBLIC_APP_URL ? `${process.env.NEXT_PUBLIC_APP_URL}/api/health` : '/api/health';
      const res = await fetch(url, { cache: 'no-store' });
      if (!res.ok && res.status !== 503) throw new Error('Failed to load health');
      return res.json();
    }

    function formatTime(iso: string): string {
      const d = new Date(iso);
      return d.toLocaleTimeString('en-US', { hour: 'numeric', minute: '2-digit', timeZoneName: 'short' });
    }

    export default async function HealthPage() {
      const data = await fetchHealth();

      const services: HealthService[] = [
        { name: 'supabase', status: data.services.supabase.status, latency: data.services.supabase.latency ?? null },
        { name: 'anthropic', status: data.services.anthropic.status, latency: data.services.anthropic.latency ?? null },
        { name: 'google_calendar', status: data.services.google_calendar.status, note: data.services.google_calendar.note ?? 'per-user OAuth' },
      ];

      return (
        <div className="relative min-h-screen bg-[var(--canvas)] px-6 py-12">
          <div className="max-w-4xl mx-auto">
            <h1 className="font-serif text-[32px] font-semibold tracking-tight text-[var(--ink)] mb-2">
              System Health
            </h1>
            <p className="font-serif text-base text-[var(--ink-muted)] mb-12">
              Service connectivity at {formatTime(data.checkedAt)}.
            </p>
            <table className="w-full">
              <thead>
                <tr className="font-mono text-[12px] tracking-[0.06em] text-[var(--ink-muted)] uppercase border-b border-[rgb(20_37_32_/_0.15)] dark:border-[rgb(255_255_255_/_0.15)]">
                  <th className="text-left py-3 pr-6 font-normal">Service</th>
                  <th className="text-left py-3 pr-6 font-normal">Status</th>
                  <th className="text-right py-3 font-normal">Latency</th>
                </tr>
              </thead>
              <tbody className="divide-y divide-[rgb(20_37_32_/_0.15)] dark:divide-[rgb(255_255_255_/_0.15)]">
                {services.map((svc) => (
                  <tr key={svc.name}>
                    <td className="py-4 font-mono text-[12px] text-[var(--ink)]">{svc.name}</td>
                    <td className={cn(
                      'py-4 font-mono text-[12px]',
                      svc.status === 'ok' && 'text-[var(--ink-sage)]',

                      {svc.latency != null ? `${svc.latency}ms` : svc.note ?? '—'}

                  </tr>
                ))}
              </tbody>
            </table>
          </div>
        </div>
      );
    }
