    ```typescript
    "use client";

    import { useEffect, useState, useCallback } from "react";
    import {
      DEFAULT_VOICE_SETTINGS,
      VOICE_SETTINGS_KEY,
    } from "@/lib/voice/constants";
    import type { VoiceSettings } from "@/lib/voice/types";

    /**
     * Phase 7 Plan 07-02 — voice settings hook.
     *
     * Single-user MVP: state lives in localStorage. ThemeToggle's mount-guard
     * pattern is replicated to prevent SSR hydration mismatch (server cannot
     * see localStorage; render DEFAULT_VOICE_SETTINGS until the useEffect
     * fires post-mount).
     *
     * Plan 07-03 (JarvisListener) and Plan 07-04 (TTS pipeline) consume this
     * hook. The setters always persist immediately — no debounce, no batching;
     * single-user changes are rare.
     */
    export function useVoiceSettings() {
      const [settings, setSettings] = useState<VoiceSettings>(DEFAULT_VOICE_SETTINGS);
      const [mounted, setMounted] = useState(false);

      useEffect(() => {
        try {
          const raw = window.localStorage.getItem(VOICE_SETTINGS_KEY);
          if (raw) {
            const parsed = JSON.parse(raw) as Partial<VoiceSettings>;
            setSettings({ ...DEFAULT_VOICE_SETTINGS, ...parsed });
          }
        } catch (err) {
          console.warn("[voice-settings] malformed localStorage; using defaults", err);
        }
        setMounted(true);
      }, []);

      const update = useCallback((patch: Partial<VoiceSettings>) => {
        setSettings((prev) => {
          const next = { ...prev, ...patch };
          try {
            window.localStorage.setItem(VOICE_SETTINGS_KEY, JSON.stringify(next));
          } catch (err) {
            console.warn("[voice-settings] write failed", err);
          }
          return next;
        });
      }, []);

      return { settings, mounted, update };
    }
    ```

    ```typescript
    "use client";

    import { useState, useRef, useEffect } from "react";
    import { Dialog, DialogContent, DialogHeader, DialogTitle, DialogDescription } from "@/components/ui/dialog";
    import { Button } from "@/components/ui/button";
    import {
      AUDIO_CONSTRAINTS,
      DEFAULT_VOICE_ID,
      WELCOME_GREETING,
    } from "@/lib/voice/constants";

    interface Props {
      open: boolean;
      onEnabled: (params: { deviceId: string | null; voiceId: string; audioContext: AudioContext }) => void;
      onCancel: () => void;
      hasHeardWelcome: boolean;
    }

    /** Three audition voice IDs — Posh (front-runner per D-03), George, AK shortlist. */
    const AUDITION_VOICES = [
      { id: "JBFqnCBsd6RMkjVDRZzb", name: "George", desc: "Warm, articulate British male" },
      { id: "EXAVITQu4vr4xnSDxMaL", name: "Posh", desc: "Theatrical butler — canon-fit (front-runner)" }, // canonical Posh voice ID
      { id: "ThT5KcBeYPX3keUQqHPh", name: "Dorothy", desc: "Wild-card option" },
    ];
    const AUDITION_LINE = "Very good, sir. Shall we proceed?";

    export function EnableVoiceModal({ open, onEnabled, onCancel, hasHeardWelcome }: Props) {
      const [stage, setStage] = useState<"intro" | "permission-denied" | "audition">("intro");
      const [selectedVoiceId, setSelectedVoiceId] = useState(DEFAULT_VOICE_ID);
      const [auditioning, setAuditioning] = useState<string | null>(null);
      const streamRef = useRef<MediaStream | null>(null);
      const audioContextRef = useRef<AudioContext | null>(null);

      // Step 1: when modal opens, request mic. This is the gesture-consuming call.
      useEffect(() => {
        if (!open || stage !== "intro") return;
        let cancelled = false;
        (async () => {
          try {
            const stream = await navigator.mediaDevices.getUserMedia({ audio: AUDIO_CONSTRAINTS });
            if (cancelled) { stream.getTracks().forEach((t) => t.stop()); return; }
            streamRef.current = stream;
            setStage("audition");
          } catch (err) {
            console.warn("[voice] mic permission denied", err);
            setStage("permission-denied");
          }
        })();
        return () => { cancelled = true; };
      }, [open, stage]);

      // Cleanup on close
      useEffect(() => {
        if (!open && streamRef.current) {
          streamRef.current.getTracks().forEach((t) => t.stop());
          streamRef.current = null;
        }
      }, [open]);

      // Audition a voice — HTTP fetch /api/jarvis/tts, play in transient AudioContext.
      async function handleAudition(voiceId: string) {
        setAuditioning(voiceId);
        try {
          const res = await fetch("/api/jarvis/tts", {
            method: "POST",
            headers: { "Content-Type": "application/json" },
            body: JSON.stringify({ text: AUDITION_LINE, voiceId }),
          });
          if (!res.ok) throw new Error(`audition failed ${res.status}`);
          const buf = await res.arrayBuffer();
          // Transient context just for audition; the real one is created at Enable click
          const ctx = new AudioContext();
          const decoded = await ctx.decodeAudioData(buf);
          const src = ctx.createBufferSource();
          src.buffer = decoded;
          src.connect(ctx.destination);
          src.start();
          src.onended = () => { ctx.close(); setAuditioning(null); };
        } catch (err) {
          console.error("[voice] audition failed", err);
          setAuditioning(null);
        }
      }

      // THE CRITICAL CLICK HANDLER — AudioContext unlock + welcome greeting in one user-gesture frame.
      async function handleEnableClick() {
        // 1. Create + resume AudioContext SYNCHRONOUSLY in the click handler (Pattern 6 / CRITICAL_PHASE7_CONCERNS #1)
        const audioContext = new AudioContext();
        if (audioContext.state === "suspended") {
          await audioContext.resume();
        }
        audioContextRef.current = audioContext;

        // 2. Capture chosen mic deviceId via enumerateDevices (now permitted post-grant)
        let deviceId: string | null = null;
        try {
          const devices = await navigator.mediaDevices.enumerateDevices();
          const firstInput = devices.find((d) => d.kind === "audioinput");
          deviceId = firstInput?.deviceId ?? null;
        } catch (err) {
          console.warn("[voice] enumerateDevices failed", err);
        }

        // 3. One-shot welcome greeting on first enable (D-03 / CRITICAL_PHASE7_CONCERNS #7)
        if (!hasHeardWelcome) {
          try {
            const res = await fetch("/api/jarvis/tts", {
              method: "POST",
              headers: { "Content-Type": "application/json" },
              body: JSON.stringify({ text: WELCOME_GREETING, voiceId: selectedVoiceId }),
            });
            if (res.ok) {
              const buf = await res.arrayBuffer();
              const decoded = await audioContext.decodeAudioData(buf);
              const src = audioContext.createBufferSource();
              src.buffer = decoded;
              src.connect(audioContext.destination);
              src.start();
            }
          } catch (err) {
            console.warn("[voice] welcome greeting failed", err);
          }
        }

        // 4. Hand control back to caller (Settings page) — it will flip voiceEnabled + persist deviceId+voiceId+hasHeardWelcome
        onEnabled({ deviceId, voiceId: selectedVoiceId, audioContext });
      }

      return (
        <Dialog open={open} onOpenChange={(o) => !o && onCancel()}>
          <DialogContent>
            <DialogHeader>
              <DialogTitle>Enable JARVIS Voice</DialogTitle>
              <DialogDescription>
                {stage === "intro" && "Requesting microphone access..."}
                {stage === "permission-denied" && "Microphone access denied. To enable, open your browser's site settings and allow microphone for this domain, then try again."}
                {stage === "audition" && "Choose a voice. Tap to audition."}
              </DialogDescription>
            </DialogHeader>

            {stage === "audition" && (
              <div className="space-y-2">
                {AUDITION_VOICES.map((v) => (
                  <button
                    key={v.id}
                    type="button"
                    onClick={() => { setSelectedVoiceId(v.id); handleAudition(v.id); }}
                    aria-pressed={selectedVoiceId === v.id}
                    className="..."
                  >
                    {v.name} — {v.desc} {auditioning === v.id && "(playing...)"}
                  </button>
                ))}
                <Button type="button" onClick={handleEnableClick}>Enable</Button>
              </div>
            )}

            {stage === "permission-denied" && (
              <Button type="button" onClick={() => setStage("intro")}>Try again</Button>
            )}
          </DialogContent>
        </Dialog>
      );
    }
    ```
