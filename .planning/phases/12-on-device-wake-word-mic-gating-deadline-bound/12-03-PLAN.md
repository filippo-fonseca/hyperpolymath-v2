        if (!raw) return DEFAULT_VOICE_SETTINGS;

          discreetMode?: boolean;     // legacy (Phase 7)

        // Phase 12 WAKE-05 forward migration (RESEARCH §Pattern 7).

        const listeningMode: ListeningMode =
          parsed.listeningMode ??
          (parsed.discreetMode ? "discreet" : "wake-word");

        // Drop legacy keys explicitly so subsequent JSON.stringify omits them.
        const { discreetMode: _legacyDiscreet, wakeWordPhrase: _legacyPhrase, ...clean } = parsed;

        return {
          ...DEFAULT_VOICE_SETTINGS,
          ...clean,
          listeningMode,
        };
      } catch (err) {
        console.warn("[voice-settings] malformed localStorage entry; using defaults", err);
        return DEFAULT_VOICE_SETTINGS;

    }

    import { useRef } from "react";
    import { MicOff, Mic } from "lucide-react";
    import { useVoiceSettings } from "@/lib/voice/use-voice-settings";
    import type { ListeningMode } from "@/lib/voice/types";
    import { cn } from "@/lib/utils";
    import { Tooltip, TooltipContent, TooltipTrigger } from "@/components/ui/tooltip";

    /**
     * Phase 12 Plan 12-03 — header Discreet quick toggle (D-02).
     *
     * Behavior (D-01 mode matrix):
     *   - In any non-discreet mode: clicking switches to 'discreet' and remembers the previous mode.
     *   - In 'discreet' mode: clicking restores the previously remembered mode (default 'wake-word').
     *
     * Visual: filled cyan + MicOff icon when in 'discreet'; outline + Mic icon otherwise.
     * Cmd+Shift+J PTT works in ALL three modes — this toggle does NOT affect PTT.
     */
    export function DiscreetToggleButton() {
      const { settings, mounted, update } = useVoiceSettings();
      // Remember the user's last non-discreet mode so toggling back restores it.
      // Default 'wake-word' for cold start when the button is mounted while already discreet.
      const previousModeRef = useRef<ListeningMode>(
        settings.listeningMode !== "discreet" ? settings.listeningMode : "wake-word",
      );

      if (!mounted || !settings.voiceEnabled) return null;
      const inDiscreet = settings.listeningMode === "discreet";

      function handleClick() {
        if (inDiscreet) {
          update({ listeningMode: previousModeRef.current });
        } else {
          previousModeRef.current = settings.listeningMode;
          update({ listeningMode: "discreet" });
        }
      }

      return (
        <Tooltip>
          <TooltipTrigger asChild>
            <button
              type="button"
              onClick={handleClick}
              aria-pressed={inDiscreet}
              aria-label={inDiscreet ? "Exit Discreet mode" : "Enter Discreet mode"}
              className={cn(
                "h-9 w-9 inline-flex items-center justify-center rounded-md transition-colors duration-150 ease-out cursor-pointer-always",
                inDiscreet
                  ? "border border-[var(--hud-cyan)] bg-[var(--surface-raised)] text-[var(--hud-cyan)]"
                  : "border border-transparent hover:border-[var(--edge)] text-[var(--ink-muted)]",
              )}
            >
              {inDiscreet ? <MicOff size={16} strokeWidth={1.5} /> : <Mic size={16} strokeWidth={1.5} />}
            </button>
          </TooltipTrigger>
          <TooltipContent>
            {inDiscreet

          </TooltipContent>
        </Tooltip>

    }
