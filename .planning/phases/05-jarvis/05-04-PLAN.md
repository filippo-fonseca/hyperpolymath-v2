```typescript
"use server";

import { z } from "zod";
import { db } from "@/lib/db";
import { captures, tasks } from "@/lib/db/schema";
import { createClient } from "@/lib/supabase/server";
import { and, eq } from "drizzle-orm";
import { getValidGcalToken, GcalTokenRevokedError, GcalNotConnectedError } from "@/lib/gcal/token";
import { createOAuth2Client } from "@/lib/gcal/client";
import { deleteEvent } from "@/lib/gcal/events";
import { google } from "googleapis";

// existing convertCaptureToTask from Plan 05-02 stays...

const UndoTargetSchema = z.discriminatedUnion("kind", [
  z.object({ kind: z.literal("task"), id: z.string().uuid() }),
  z.object({ kind: z.literal("capture"), id: z.string().uuid() }),
  z.object({ kind: z.literal("event"), id: z.string().min(1), calendarId: z.string().min(1) }),
]);

export async function undoJarvisAction(target: z.infer<typeof UndoTargetSchema>) {
  const supabase = await createClient();
  const { data: claims } = await supabase.auth.getClaims();
  if (!claims?.claims?.sub) return { ok: false as const, error: "Unauthorized" };
  const userId = claims.claims.sub;

  const parsed = UndoTargetSchema.safeParse(target);
  if (!parsed.success) return { ok: false as const, error: parsed.error.message };

  try {
    if (parsed.data.kind === "task") {
      // B5: Phase 2 uses HARD delete (tasks/captures have no deleted_at column).
      // CONTEXT.md D-04 wording is reconciled to "hard delete (matches Phase 2 task delete pattern)".
      const result = await db.delete(tasks)
        .where(and(eq(tasks.id, parsed.data.id), eq(tasks.userId, userId)));
      return { ok: true as const };
    }
    if (parsed.data.kind === "capture") {
      // B5: hard delete (captures table has no deleted_at column).
      await db.delete(captures)
        .where(and(eq(captures.id, parsed.data.id), eq(captures.userId, userId)));
      return { ok: true as const };
    }
    // event — gcal delete via lib/gcal/events deleteEvent helper (Phase 4 boundary)
    const accessToken = await getValidGcalToken(userId);
    const oauth2 = createOAuth2Client();
    oauth2.setCredentials({ access_token: accessToken });
    const cal = google.calendar({ version: "v3", auth: oauth2 });
    try {
      await deleteEvent(cal, parsed.data.calendarId, parsed.data.id);
    } catch (err: any) {
      // 404 = already deleted; treat as success (D-04 best-effort)
      if (err?.code !== 404) throw err;
    }
    return { ok: true as const };
  } catch (err) {
    if (err instanceof GcalTokenRevokedError || err instanceof GcalNotConnectedError) {
      return { ok: false as const, error: "Calendar disconnected" };
    }
    return { ok: false as const, error: String(err) };
  }
}
```

```typescript
"use client";

import { useCallback, useEffect, useRef, useState } from "react";

export function useUndoCountdown(initialSeconds: number, onExpire: () => void): { seconds: number; cancel: () => void } {
  const [seconds, setSeconds] = useState(initialSeconds);
  const expireRef = useRef(onExpire);
  expireRef.current = onExpire;
  const intervalRef = useRef<ReturnType<typeof setInterval> | null>(null);

  useEffect(() => {
    intervalRef.current = setInterval(() => {
      setSeconds((s) => {
        if (s <= 1) {
          if (intervalRef.current) clearInterval(intervalRef.current);
          intervalRef.current = null;
          expireRef.current();
          return 0;
        }
        return s - 1;
      });
    }, 1000);
    return () => {
      if (intervalRef.current) clearInterval(intervalRef.current);
    };
  }, []);

  const cancel = useCallback(() => {
    if (intervalRef.current) {
      clearInterval(intervalRef.current);
      intervalRef.current = null;
    }
  }, []);

  return { seconds, cancel };
}
```

```typescript
const handleUndo = useCallback(async (assistantTurnId: string, toolUseId: string, target: UndoTarget) => {
  // Optimistic local update
  setTurns((prev) => prev.map((t) =>
    t.id === assistantTurnId && t.kind === "assistant"
      ? { ...t, actions: t.actions.map((a) => a.toolUseId === toolUseId ? { ...a, undone: true } : a) }

      t.id === assistantTurnId && t.kind === "assistant"
        ? { ...t, actions: t.actions.map((a) => a.toolUseId === toolUseId ? { ...a, undone: false } : a) }
        : t
    ));
  } else {
```

```typescript
"use client";

import { useState } from "react";
import { Dialog, DialogContent, DialogHeader, DialogTitle, DialogFooter } from "@/components/ui/dialog";
import { Input } from "@/components/ui/input";
import { Select, SelectContent, SelectItem, SelectTrigger, SelectValue } from "@/components/ui/select";
import { Button } from "@/components/ui/button";
import { Label } from "@/components/ui/label";
import { toast } from "sonner";
import { ProjectMultiSelect } from "@/components/shared/ProjectMultiSelect";
import { convertCaptureToTask } from "@/app/actions/jarvis";

interface Props {
  open: boolean;
  onOpenChange: (open: boolean) => void;
  capture: { id: string; content: string };
  existingProjectIds: string[];
  availableProjects: Array<{ id: string; name: string; icon?: string | null }>;
}

export function ConvertCaptureToTaskDialog({ open, onOpenChange, capture, existingProjectIds, availableProjects }: Props) {
  const [title, setTitle] = useState(() => capture.content.slice(0, 80));
  const [priority, setPriority] = useState<"P∞" | "P1" | "P2" | "P3">("P3");
  const [projectIds, setProjectIds] = useState<string[]>(existingProjectIds);
  const [pending, setPending] = useState(false);

  const handleSubmit = async () => {
    setPending(true);
    const result = await convertCaptureToTask({ captureId: capture.id, title, priority, projectIds });
    setPending(false);
    if (result.ok) {
      toast.success("Converted to task");
      onOpenChange(false);
    } else {
      toast.error(`Failed to convert: ${result.error}`);
    }
  };

  return (
    <Dialog open={open} onOpenChange={onOpenChange}>
      <DialogContent>
        <DialogHeader>
          <DialogTitle>Convert capture to task</DialogTitle>
        </DialogHeader>
        <div className="space-y-4">
          <div>
            <Label htmlFor="convert-title">Task title</Label>
            <Input id="convert-title" value={title} onChange={(e) => setTitle(e.target.value)} maxLength={500} />
          </div>
          <div>
            <Label>Priority</Label>
            <Select value={priority} onValueChange={(v) => setPriority(v as any)}>
              <SelectTrigger><SelectValue /></SelectTrigger>
              <SelectContent>
                <SelectItem value="P∞">P∞ (top)</SelectItem>
                <SelectItem value="P1">P1</SelectItem>
                <SelectItem value="P2">P2</SelectItem>
                <SelectItem value="P3">P3</SelectItem>
              </SelectContent>
            </Select>
          </div>
          <div>
            <Label>Projects</Label>
            <ProjectMultiSelect availableProjects={availableProjects} selectedIds={projectIds} onChange={setProjectIds} />
          </div>
        </div>
        <DialogFooter>
          <Button variant="ghost" onClick={() => onOpenChange(false)} disabled={pending}>Cancel</Button>
          <Button onClick={handleSubmit} disabled={pending || title.length === 0}>
            {pending ? "Converting..." : "Convert"}
          </Button>
        </DialogFooter>
      </DialogContent>
    </Dialog>
  );
}
```

**JARVIS-15 latency telemetry — `lib/jarvis/latency-check.ts`:**
- Small CLI helper used by the verification smoke (Task 4) to query `jarvis_events` for p50 + p95 first_token_ms and latency_ms across the smoke session.
- Signature: `getLatencyStats(userId: string, sinceMinutes: number): Promise<{ count: number; first_token_p50: number; first_token_p95: number; latency_p50: number; latency_p95: number }>`.
- Query: `SELECT first_token_ms, latency_ms FROM jarvis_events WHERE user_id = $1 AND created_at > NOW() - INTERVAL '$2 minutes' ORDER BY first_token_ms`.
- Computes percentiles in TS (small N — no PG percentile_cont needed).

```typescript
import { db } from "@/lib/db";
import { jarvisEvents } from "@/lib/db/schema";
import { and, eq, gt, isNotNull } from "drizzle-orm";

export interface LatencyStats {
  count: number;
  first_token_p50: number;
  first_token_p95: number;
  latency_p50: number;
  latency_p95: number;
}

function percentile(sorted: number[], p: number): number {
  if (sorted.length === 0) return 0;
  const idx = Math.ceil(sorted.length * p) - 1;
  return sorted[Math.max(0, Math.min(idx, sorted.length - 1))];
}

export async function getLatencyStats(userId: string, sinceMinutes: number): Promise<LatencyStats> {
  const sinceDate = new Date(Date.now() - sinceMinutes * 60 * 1000);
  const rows = await db
    .select({ firstTokenMs: jarvisEvents.firstTokenMs, latencyMs: jarvisEvents.latencyMs })
    .from(jarvisEvents)
    .where(and(eq(jarvisEvents.userId, userId), gt(jarvisEvents.createdAt, sinceDate), isNotNull(jarvisEvents.firstTokenMs)));
  const ft = rows.map((r) => r.firstTokenMs ?? 0).sort((a, b) => a - b);
  const lat = rows.map((r) => r.latencyMs ?? 0).sort((a, b) => a - b);
  return {
    count: rows.length,
    first_token_p50: percentile(ft, 0.5),
    first_token_p95: percentile(ft, 0.95),
    latency_p50: percentile(lat, 0.5),
    latency_p95: percentile(lat, 0.95),
  };
}
```
