**streamJarvis(payload, callbacks, signal):**
- Signature: `streamJarvis(payload: JarvisRequest, callbacks: { onText, onAction, onDone, onError }, signal?: AbortSignal): Promise<void>`.
- POSTs to `/api/jarvis` with `Content-Type: application/json`, `X-Voice-Active: "false"` header, JSON body, passes through `signal`.
- Reads response body via `response.body.pipeThrough(new TextDecoderStream()).getReader()` per research §3.2 (NOT EventSource — POST unsupported by EventSource spec).
- Parses SSE chunks split on `\n\n`. Each chunk has `event: X` + `data: <JSON>` lines.
- Routes each event to the matching callback:
  - `event: text` → `onText(data.delta)`.
  - `event: action` → `onAction(data)`.
  - `event: done` → `onDone(data.usage)`.
  - `event: error` → `onError(data.message)`.
- Stops cleanly on `signal.abort()` or stream end.
- Test cases:
  1. Stream emits 1 text + 1 action + 1 done → callbacks fire in order.
  2. Stream emits 3 actions (parallel) + 1 done → onAction fires 3x.
  3. Stream emits error event → onError fires, no onDone.
  4. Signal aborts mid-stream → reader cancels, onError fires with abort message.

**apps/web/components/jarvis/jarvis-stream-client.ts** — per research §3.2:

```typescript
import type { ParsedDate } from "@hyperpolymath/jarvis-core";

export interface JarvisRequest {
  input: string;
  history: Array<{ role: "user" | "assistant"; content: string }>;
  parsedDates?: ParsedDate[];
  slashCommand?: "task" | "capture" | "event" | "help" | null;
  linkedProjectIds?: string[];           // M5: client-validated project UUIDs
  linkedHashtags?: string[];             // M6: client-extracted hashtag names
}

export interface JarvisCallbacks {
  onText: (delta: string) => void;
  onAction: (data: { toolUseId: string; name: string; result: { ok: boolean; id?: string; receipt?: Record<string, unknown>; error?: string; kind?: string } }) => void;
  onDone: (usage: Record<string, number>) => void;
  onError: (message: string) => void;
}

export async function streamJarvis(payload: JarvisRequest, callbacks: JarvisCallbacks, signal?: AbortSignal): Promise<void> {
  let response: Response;
  try {
    response = await fetch("/api/jarvis", {
      method: "POST",
      headers: {
        "Content-Type": "application/json",
        "X-Voice-Active": "false",                       // Phase 7 will toggle
      },
      body: JSON.stringify(payload),
      signal,
    });
  } catch (err: any) {
    callbacks.onError(err?.name === "AbortError" ? "aborted" : String(err?.message ?? err));
    return;
  }

  if (!response.ok || !response.body) {
    callbacks.onError(`HTTP ${response.status}`);
    return;
  }

  const reader = response.body.pipeThrough(new TextDecoderStream()).getReader();
  let buffer = "";

  try {
    while (true) {
      const { done, value } = await reader.read();
      if (done) break;
      buffer += value;
      let idx = buffer.indexOf("\n\n");
      while (idx !== -1) {
        const chunk = buffer.slice(0, idx);
        buffer = buffer.slice(idx + 2);
        idx = buffer.indexOf("\n\n");
        const eventName = chunk.match(/^event: (\w+)$/m)?.[1] ?? "";
        const dataMatch = chunk.match(/^data: (.+)$/m)?.[1] ?? "null";
        let data: any;
        try { data = JSON.parse(dataMatch); } catch { continue; }
        if (eventName === "text") callbacks.onText(data.delta ?? "");
        else if (eventName === "action") callbacks.onAction(data);
        else if (eventName === "done") callbacks.onDone(data.usage ?? {});
        else if (eventName === "error") callbacks.onError(data.message ?? "Unknown error");
      }
    }
  } catch (err: any) {
    if (err?.name !== "AbortError") callbacks.onError(String(err?.message ?? err));
  } finally {
    try { await reader.cancel(); } catch {}
  }
}
```

**apps/web/components/jarvis/ProjectSuggestionList.tsx** — mirror `HashtagSuggestionList.tsx` (Phase 2 commit 25e5e57). forwardRef + useImperativeHandle so the suggestion popover handles arrow keys + click + Enter despite being mounted outside the editor's React tree (the canonical TipTap pattern documented in `apps/web/components/captures/HashtagSuggestionList.tsx`'s comment block).

```typescript
"use client";

import { forwardRef, useEffect, useImperativeHandle, useState } from "react";
import { cn } from "@/lib/utils";
import type { ProjectOption } from "./project-suggestions";

interface Props {
  items: ProjectOption[];
  command: (item: ProjectOption) => void;
}

export interface ProjectSuggestionListHandle {
  onKeyDown: (props: { event: KeyboardEvent }) => boolean;
}

export const ProjectSuggestionList = forwardRef<ProjectSuggestionListHandle, Props>(({ items, command }, ref) => {
  const [selectedIndex, setSelectedIndex] = useState(0);

  useEffect(() => setSelectedIndex(0), [items]);

  const select = (idx: number) => {
    const item = items[idx];
    if (item) command(item);
  };

  useImperativeHandle(ref, () => ({
    onKeyDown: ({ event }) => {
      if (event.key === "ArrowUp") {
        setSelectedIndex((i) => (i + items.length - 1) % items.length);
        return true;
      }
      if (event.key === "ArrowDown") {
        setSelectedIndex((i) => (i + 1) % items.length);
        return true;
      }
      if (event.key === "Enter") {
        select(selectedIndex);
        return true;
      }
      return false;
    },
  }));

  if (items.length === 0) return null;

  return (
    <div className="z-50 min-w-[12rem] rounded border bg-popover p-1 shadow-md font-mono text-sm">
      {items.map((item, idx) => (
        <button
          key={item.id ?? `idx-${idx}`}
          type="button"
          onClick={() => select(idx)}
          onMouseEnter={() => setSelectedIndex(idx)}
          className={cn(
            "flex w-full items-center gap-2 rounded px-2 py-1 text-left",
            idx === selectedIndex ? "bg-accent text-accent-foreground" : "hover:bg-accent/50",
          )}
        >
          {item.icon ? <span className="text-base">{item.icon}</span> : null}
          <span>{item.label ?? item.id}</span>
        </button>
      ))}
    </div>
  );
});

ProjectSuggestionList.displayName = "ProjectSuggestionList";
```

**apps/web/components/jarvis/project-suggestions.ts** — clone shape of `apps/web/components/captures/tiptap-suggestions.ts` with `char: "$"`. Critical: this is a **separate Mention extension** from the hashtag one — different node name (`projectMention`) so the two siblings coexist in the TipTap editor without trigger collision.

```typescript
import { createRoot, type Root } from "react-dom/client";
import { createElement, createRef, type RefObject } from "react";
import type { SuggestionOptions } from "@tiptap/suggestion";
import type { MentionNodeAttrs } from "@tiptap/extension-mention";
import { ProjectSuggestionList, type ProjectSuggestionListHandle } from "./ProjectSuggestionList";

export interface ProjectOption extends MentionNodeAttrs {
  icon?: string | null;
}

interface ProjectSource {
  id: string;       // canonical project UUID
  name: string;     // user-facing project name (case preserved)
  icon?: string | null;
}

export function createProjectSuggestion(getProjects: () => ProjectSource[]): Omit<SuggestionOptions<ProjectOption, ProjectOption>, "editor"> {
  return {
    char: "$",
    startOfLine: false,
    allowSpaces: false,

    items: ({ query }) => {
      const projects = getProjects();
      const lower = query.toLowerCase();
      return projects
        .filter((p) => p.name.toLowerCase().includes(lower))
        .slice(0, 8)
        .map((p) => ({ id: p.id, label: p.name, icon: p.icon ?? null }));
    },

    command: ({ editor, range, props }) => {
      const projectId = typeof props.id === "string" ? props.id : "";
      const projectName = props.label ?? "";
      if (!projectId) return;
      editor
        .chain()
        .focus()
        .insertContentAt(range, [
          { type: "projectMention", attrs: { id: projectId, label: projectName } },
          { type: "text", text: " " },
        ])
        .run();
    },

    render: () => {
      let root: Root | null = null;
      let container: HTMLDivElement | null = null;
      const componentRef: RefObject<ProjectSuggestionListHandle | null> = createRef();

      return {
        onStart: (props) => {
          container = document.createElement("div");
          container.style.position = "absolute";
          container.style.zIndex = "9999";
          document.body.appendChild(container);
          root = createRoot(container);
          const rect = props.clientRect?.();
          if (rect) {
            container.style.top = `${rect.bottom + window.scrollY}px`;
            container.style.left = `${rect.left + window.scrollX}px`;
          }
          root.render(createElement(ProjectSuggestionList, { items: props.items as ProjectOption[], command: props.command, ref: componentRef }));
        },
        onUpdate: (props) => {
          if (!root || !container) return;
          const rect = props.clientRect?.();
          if (rect) {
            container.style.top = `${rect.bottom + window.scrollY}px`;
            container.style.left = `${rect.left + window.scrollX}px`;
          }
          root.render(createElement(ProjectSuggestionList, { items: props.items as ProjectOption[], command: props.command, ref: componentRef }));
        },
        onKeyDown: (props) => componentRef.current?.onKeyDown(props) ?? false,
        onExit: () => {
          root?.unmount();
          if (container) document.body.removeChild(container);
          root = null;
          container = null;
        },
      };
    },
  };
}
```

```typescript
"use client";

import { useEditor, EditorContent } from "@tiptap/react";
import StarterKit from "@tiptap/starter-kit";
import Mention from "@tiptap/extension-mention";
import { useCallback, useEffect, useRef, useState } from "react";
import { createHashtagSuggestion } from "@/components/captures/tiptap-suggestions";
import { createProjectSuggestion, type ProjectOption } from "./project-suggestions";
import { SlashCommandPopover, type SlashCommandKey } from "./SlashCommandPopover";
import { parseDates, parsePriority, parseSlashCommand, type ParsedDate } from "@hyperpolymath/jarvis-core";

interface ProjectSource { id: string; name: string; icon?: string | null }
interface HashtagSource { id: string; name: string; displayName: string }

interface Props {
  userTimezone: string;
  getProjects: () => ProjectSource[];
  getHashtags: () => HashtagSource[];
  onSubmit: (payload: {
    input: string;
    parsedDates: ParsedDate[];
    slashCommand: SlashCommandKey | null;
    hashtags: string[];
    projectIds: string[];
  }) => void;
  disabled?: boolean;
}

export function JarvisInput({ userTimezone, getProjects, getHashtags, onSubmit, disabled }: Props) {
  const [slashOpen, setSlashOpen] = useState(false);
  const [slashQuery, setSlashQuery] = useState("");
  const [slashSelected, setSlashSelected] = useState(0);

  // Build the project Mention as an extended node so it has a DIFFERENT node name
  // from the default hashtag Mention — both can coexist.
  const ProjectMention = Mention.extend({ name: "projectMention" });

  const editor = useEditor({
    extensions: [
      StarterKit.configure({ heading: false, codeBlock: false, blockquote: false, horizontalRule: false }),
      Mention.configure({
        HTMLAttributes: { class: "hashtag-chip-inline" },
        suggestion: createHashtagSuggestion(getHashtags),
      }),
      ProjectMention.configure({
        HTMLAttributes: { class: "project-chip-inline" },
        suggestion: createProjectSuggestion(getProjects),
      }),
    ],
    immediatelyRender: false,
    editorProps: {
      handleKeyDown: (_view, event) => {
        // N4: read text from the ProseMirror view (not closure-captured `editor`,
        // which is used-before-definition inside useEditor's editorProps).
        const text = _view.state.doc.textContent;

        // M3: If the slash popover is open, intercept Arrow/Enter/Esc before submit.
        if (slashOpen) {
          if (event.key === "ArrowUp") {
            setSlashSelected((i) => Math.max(0, i - 1));
            return true;
          }
          if (event.key === "ArrowDown") {
            setSlashSelected((i) => i + 1);   // SlashCommandPopover clamps via filtered list
            return true;
          }
          if (event.key === "Enter") {
            // Let the popover's onSelect fire by simulating the click of the selected item.
            // Implementation detail: SlashCommandPopover is presentational; JarvisInput owns
            // the selection state. On Enter we re-derive the chosen command from slashQuery
            // and slashSelected, replace the editor content with `/<command> `, and submit.
            const COMMANDS = ["task", "capture", "event", "help"] as const;
            const filtered = COMMANDS.filter((c) => c.startsWith(slashQuery.toLowerCase()));
            const picked = filtered[Math.min(slashSelected, filtered.length - 1)];
            if (picked && picked !== "help") {
              // Pre-pend the chosen slash command + the rest of the body and submit.
              const bodyAfterSlash = text.trimStart().replace(/^\/\S*\s*/, "");
              editor?.commands.setContent(`/${picked} ${bodyAfterSlash}`);
              setSlashOpen(false);
              handleSubmit();
              return true;
            }
            if (picked === "help") {
              setSlashOpen(false);
              return true;
            }
            return false;
          }
          if (event.key === "Escape") {
            setSlashOpen(false);
            return true;
          }
        }

        if (event.key === "Enter" && !event.shiftKey) {
          handleSubmit();
          return true;
        }
        return false;
      },
    },
    onUpdate: ({ editor }) => {
      const text = editor.getText().trimStart();
      if (text.startsWith("/")) {
        const slashBody = text.slice(1).split(/\s/)[0];
        setSlashOpen(true);
        setSlashQuery(slashBody);
      } else {
        setSlashOpen(false);
      }
    },
  });

  const handleSubmit = useCallback(() => {
    if (!editor) return;
    const text = editor.getText().trim();
    if (text.length === 0) return;

    // Parse slash command
    const slashParsed = parseSlashCommand(text);
    const inputText = slashParsed ? slashParsed.body : text;
    const slashCommand = slashParsed?.command ?? null;

    // Extract mentions from editor JSON
    const json = editor.getJSON();
    const hashtags: string[] = [];
    const projectIds: string[] = [];
    function walk(node: any) {
      if (!node) return;
      if (node.type === "mention" && node.attrs?.label) hashtags.push(String(node.attrs.label).toLowerCase());
      if (node.type === "projectMention" && node.attrs?.id) projectIds.push(String(node.attrs.id));
      if (Array.isArray(node.content)) for (const c of node.content) walk(c);
    }
    walk(json);

    // Permissive #hashtag regex parse (Phase 2 permissive pattern)
    const hashRegex = /#([a-z0-9_-]+)/gi;
    let m: RegExpExecArray | null;
    while ((m = hashRegex.exec(inputText)) !== null) {
      const tag = m[1].toLowerCase();
      if (!hashtags.includes(tag)) hashtags.push(tag);
    }

    // Client-side date pre-parse (D-10)
    const parsedDates = parseDates(inputText, userTimezone);

    onSubmit({
      input: inputText,
      parsedDates,
      slashCommand: slashCommand === "help" ? null : slashCommand,
      hashtags,
      projectIds,
    });
    editor.commands.clearContent();
    setSlashOpen(false);
  }, [editor, userTimezone, onSubmit]);

  return (
    <div className="relative">
      <EditorContent editor={editor} className="jarvis-input rounded border bg-card px-3 py-2 font-mono text-base" />
      {slashOpen ? (
        <SlashCommandPopover
          query={slashQuery}
          selectedIndex={slashSelected}
          onSelect={(key) => {
            if (key === "help") {
              // /help is local — render command list inline; don't submit
              setSlashOpen(false);
              return;
            }
            // Otherwise, replace the slash prefix with the command form
            setSlashOpen(false);
          }}
        />
      ) : null}
    </div>
  );
}
```

```typescript
"use client";

import { useCallback, useRef, useState } from "react";
import { streamJarvis, type JarvisRequest } from "./jarvis-stream-client";
// IDs use native crypto.randomUUID() — no import needed; matches Phase 2 pattern.
import { JarvisScrollback } from "./JarvisScrollback";
import { JarvisInput } from "./JarvisInput";
import type { ScrollbackTurn, ScrollbackAction } from "./jarvis-types";

interface ProjectSource { id: string; name: string; icon?: string | null }
interface HashtagSource { id: string; name: string; displayName: string }

interface Props {
  userTimezone: string;
  initialProjects: ProjectSource[];
  initialHashtags: HashtagSource[];
}

export function JarvisConsole({ userTimezone, initialProjects, initialHashtags }: Props) {
  const [turns, setTurns] = useState<ScrollbackTurn[]>([]);
  const [streaming, setStreaming] = useState(false);
  const abortRef = useRef<AbortController | null>(null);

  // Session memory (D-06): build from scrollback. Last ~10 turns.
  const sessionHistory = useCallback(() => {
    return turns
      .slice(-10)
      .map((t) => ({
        role: t.kind === "user" ? ("user" as const) : ("assistant" as const),
        content: t.kind === "user" ? t.text : t.textDelta,
      }))
      .filter((m) => m.content.length > 0);
  }, [turns]);

  const handleSubmit = useCallback(async (payload: {
    input: string;
    parsedDates: any[];
    slashCommand: any;
    hashtags: string[];
    projectIds: string[];
  }) => {
    const userTurn: ScrollbackTurn = {
      kind: "user",
      id: crypto.randomUUID(),
      text: payload.input,
      createdAt: new Date(),
    };
    const assistantId = crypto.randomUUID();
    const assistantTurn: ScrollbackTurn = {
      kind: "assistant",
      id: assistantId,
      textDelta: "",
      actions: [],
      createdAt: new Date(),
      status: "streaming",
    };
    setTurns((prev) => [...prev, userTurn, assistantTurn]);

    const history = sessionHistory();
    setStreaming(true);
    const ac = new AbortController();
    abortRef.current = ac;

    await streamJarvis(
      {
        input: payload.input,
        history,
        parsedDates: payload.parsedDates,
        slashCommand: payload.slashCommand,
        linkedProjectIds: payload.projectIds,            // M5: forward client-validated $ chip IDs
        linkedHashtags: payload.hashtags,                // M6: forward client-extracted # tags
      } as JarvisRequest,
      {
        onText: (delta) => {
          setTurns((prev) => prev.map((t) =>
            t.id === assistantId && t.kind === "assistant"
              ? { ...t, textDelta: t.textDelta + delta }
              : t
          ));
        },
        onAction: (data) => {
          const action: ScrollbackAction = {
            toolUseId: data.toolUseId,
            name: data.name as ScrollbackAction["name"],
            result: data.result as ScrollbackAction["result"],
          };
          setTurns((prev) => prev.map((t) =>
            t.id === assistantId && t.kind === "assistant"
              ? { ...t, actions: [...t.actions, action] }
              : t
          ));
        },
        onDone: (_usage) => {
          setTurns((prev) => prev.map((t) =>
            t.id === assistantId && t.kind === "assistant"
              ? { ...t, status: "done" }
              : t
          ));
          setStreaming(false);
        },
        onError: (message) => {
          setTurns((prev) => prev.map((t) =>
            t.id === assistantId && t.kind === "assistant"
              ? { ...t, status: "error", errorMessage: message }
              : t
          ));
          setStreaming(false);
        },
      },
      ac.signal,
    );
  }, [sessionHistory]);

  return (
    <div className="flex h-[calc(100vh-3rem)] flex-col">
      <JarvisScrollback turns={turns} />
      <div className="border-t bg-card px-6 py-3">
        <JarvisInput
          userTimezone={userTimezone}
          getProjects={() => initialProjects}
          getHashtags={() => initialHashtags}
          onSubmit={handleSubmit}
          disabled={streaming}
        />
      </div>
    </div>
  );
}
```
