**validateProjectIds(userId, projectIds):**
- Returns `{ ok: true; ids: string[] }` if every id belongs to a project where `projects.user_id = userId`.
- Returns `{ ok: false; rejected: string[] }` if any id is owned by another user OR doesn't exist.
- Empty input → `{ ok: true; ids: [] }`.

**validateCalendarId(userId, calendarId):**
- Returns `{ ok: true; calendarId }` if user has gcal connection AND calendarId is in `users.gcal_visible_calendar_ids` (or matches `users.gcal_default_calendar_id`, or is the user's primary calendar).
- Returns `{ ok: false }` otherwise.
- Null input → returns `{ ok: true; calendarId: user.gcal_default_calendar_id || "primary" }`.

```typescript
import { db } from "@/lib/db";
import { projects, users } from "@/lib/db/schema";
import { eq, and, inArray } from "drizzle-orm";

export interface ValidateProjectsResult {
  ok: boolean;
  ids: string[];
  rejected: string[];
}

export async function validateProjectIds(userId: string, projectIds: string[] | undefined): Promise<ValidateProjectsResult> {
  if (!projectIds || projectIds.length === 0) return { ok: true, ids: [], rejected: [] };
  const rows = await db
    .select({ id: projects.id })
    .from(projects)
    .where(and(eq(projects.userId, userId), inArray(projects.id, projectIds)));
  const found = new Set(rows.map((r) => r.id));
  const rejected = projectIds.filter((id) => !found.has(id));
  return { ok: rejected.length === 0, ids: Array.from(found), rejected };
}

export interface ValidateCalendarResult {
  ok: boolean;
  calendarId: string | null;
}

export async function validateCalendarId(userId: string, calendarId: string | null | undefined): Promise<ValidateCalendarResult> {
  const [row] = await db
    .select({
      defaultCalendarId: users.gcalDefaultCalendarId,
      visibleCalendarIds: users.gcalVisibleCalendarIds,
    })
    .from(users)
    .where(eq(users.id, userId));
  if (!row) return { ok: false, calendarId: null };
  if (!calendarId) return { ok: true, calendarId: row.defaultCalendarId ?? "primary" };
  if (calendarId === row.defaultCalendarId) return { ok: true, calendarId };
  if (row.visibleCalendarIds && row.visibleCalendarIds.includes(calendarId)) return { ok: true, calendarId };
  if (calendarId === "primary") return { ok: true, calendarId };
  return { ok: false, calendarId: null };
}
```

```typescript
import { google } from "googleapis";
import { createOAuth2Client } from "./client";
import { getValidGcalToken } from "./token";

export interface JarvisEventInput {
  calendarId: string;
  title: string;
  description?: string;
  start: Date;
  end: Date;
}

export interface GcalEventDTO {
  id: string;
  calendarId: string;
  title: string;
  description?: string | null;
  start: string;            // ISO 8601
  end: string;              // ISO 8601
  htmlLink?: string | null;
}

export async function createEventForJarvis(
  userId: string,
  input: JarvisEventInput,
): Promise<GcalEventDTO> {
  const accessToken = await getValidGcalToken(userId);
  const oauth2 = createOAuth2Client();
  oauth2.setCredentials({ access_token: accessToken });
  const cal = google.calendar({ version: "v3", auth: oauth2 });

  const { data } = await insertEvent(cal, input.calendarId, {
    summary: input.title,
    description: input.description ?? undefined,
    start: { dateTime: input.start.toISOString() },
    end: { dateTime: input.end.toISOString() },
  });

  if (!data.id) throw new Error("gcal returned no event id");
  return {
    id: data.id,
    calendarId: input.calendarId,
    title: data.summary ?? input.title,
    description: data.description ?? null,
    start: data.start?.dateTime ?? input.start.toISOString(),
    end: data.end?.dateTime ?? input.end.toISOString(),
    htmlLink: data.htmlLink ?? null,
  };
}
```

```typescript
import { randomUUID } from "node:crypto";
import { db } from "@/lib/db";
import { tasks, captures, hashtags, capturesHashtags, capturesProjects, tasksProjects } from "@/lib/db/schema";
import { upsertHashtag } from "@/lib/db/queries/hashtags";
import { createEventForJarvis } from "@/lib/gcal/events";
import { GcalTokenRevokedError, GcalNotConnectedError } from "@/lib/gcal/token";
import { validateProjectIds, validateCalendarId } from "./validate-references";
import type { ActionExecutor, ExecutionContext, ExecutorResult, CreateTaskAction, CreateCaptureAction, CreateEventAction } from "@hyperpolymath/jarvis-core";

export function createServerExecutor(): ActionExecutor {
  return {
    async createTask(input: CreateTaskAction, ctx: ExecutionContext): Promise<ExecutorResult> {
      const projectCheck = await validateProjectIds(ctx.userId, input.project_ids);
      if (!projectCheck.ok) return { ok: false, kind: "validation", error: `Unknown projects: ${projectCheck.rejected.join(", ")}` };

      const taskId = randomUUID();
      try {
        await db.transaction(async (tx) => {
          await tx.insert(tasks).values({
            id: taskId,
            userId: ctx.userId,                            // JARVIS-12: from getClaims(), never model
            title: input.title,
            priority: input.priority ?? "P3",
            status: input.status ?? "not started",
            dueAt: input.due ? new Date(input.due) : null,
          });
          if (projectCheck.ids.length > 0) {
            await tx.insert(tasksProjects).values(projectCheck.ids.map((pid) => ({ taskId, projectId: pid })));
          }
        });
        return { ok: true, id: taskId, receipt: { id: taskId, title: input.title, priority: input.priority ?? "P3", due: input.due, project_ids: projectCheck.ids } };
      } catch (err) {
        return { ok: false, kind: "validation", error: String(err) };
      }
    },

    async createCapture(input: CreateCaptureAction, ctx: ExecutionContext): Promise<ExecutorResult> {
      const projectCheck = await validateProjectIds(ctx.userId, input.project_ids);
      if (!projectCheck.ok) return { ok: false, kind: "validation", error: `Unknown projects: ${projectCheck.rejected.join(", ")}` };

      const captureId = randomUUID();
      try {
        await db.transaction(async (tx) => {
          await tx.insert(captures).values({
            id: captureId,
            userId: ctx.userId,
            content: input.content,
            createdVia: "jarvis",                          // D-14
          });
          // Upsert hashtags (race-safe — Phase 2 pattern)
          for (const tag of input.hashtags ?? []) {
            const upserted = await upsertHashtag(ctx.userId, tag, tx);
            const hashtagId = upserted.id;
            await tx.insert(capturesHashtags).values({ captureId, hashtagId }).onConflictDoNothing();
          }
          if (projectCheck.ids.length > 0) {
            await tx.insert(capturesProjects).values(projectCheck.ids.map((pid) => ({ captureId, projectId: pid })));
          }
        });
        return { ok: true, id: captureId, receipt: { id: captureId, content: input.content, hashtags: input.hashtags ?? [], project_ids: projectCheck.ids } };
      } catch (err) {
        return { ok: false, kind: "validation", error: String(err) };
      }
    },

    async createEvent(input: CreateEventAction, ctx: ExecutionContext): Promise<ExecutorResult> {
      const calCheck = await validateCalendarId(ctx.userId, input.calendar_id);
      if (!calCheck.ok) return { ok: false, kind: "validation", error: "Calendar not found or not visible" };

      try {
        const event = await createEventForJarvis(ctx.userId, {
          calendarId: calCheck.calendarId!,
          title: input.title,
          description: input.description,
          start: new Date(input.start),
          end: new Date(input.end),
        });
        return { ok: true, id: event.id, receipt: { id: event.id, title: event.title, start: event.start, end: event.end, calendar_id: event.calendarId } };
      } catch (err) {
        if (err instanceof GcalTokenRevokedError) return { ok: false, kind: "revoked", error: "Calendar disconnected" };
        if (err instanceof GcalNotConnectedError) return { ok: false, kind: "revoked", error: "Calendar not connected" };
        return { ok: false, kind: "network", error: String(err) };
      }
    },
  };
}
```

**apps/web/app/api/jarvis/route.ts** — verbatim from research §3.1 with project-specific wiring. Critical correctness items:

1. `export const runtime = "nodejs"` — NOT "edge" (googleapis + postgres + Drizzle require Node).
2. `export const maxDuration = 60` — JARVIS-15 budget; Vercel default 300s gives headroom.
3. Headers: `Content-Type: text/event-stream`, `Cache-Control: no-cache, no-transform`, `Connection: keep-alive`, `X-Accel-Buffering: no` (load-bearing on Vercel — without it the proxy buffers SSE chunks).

6. Stream wiring uses `client.messages.stream(...).on("contentBlock", ...)` per research §1.2 — fires only on completed tool_use blocks (SDK accumulates input_json_delta internally).
7. Per completed tool_use: re-validate input with `zCreateTask` / `zCreateCapture` / `zCreateEvent` from jarvis-core (Zod 4 .parse() throws on malformed). Dispatch to executor. Enqueue SSE 'action' event.
8. After `await stream.finalMessage()`: enqueue 'done' with usage. Fire-and-forget `logJarvisEvent`.
9. On error (catch): enqueue 'error' event, swallow AbortError (client disconnect is normal).

```typescript
// apps/web/app/api/jarvis/route.ts
import { NextRequest } from "next/server";
import { createClient } from "@/lib/supabase/server";
import { getAnthropicClient, JARVIS_MODEL } from "@/lib/jarvis/anthropic-client";
import { createServerExecutor } from "@/lib/jarvis/executor";
import { logJarvisEvent } from "@/lib/jarvis/log-event";
import { db } from "@/lib/db";
import { projects, users } from "@/lib/db/schema";
import { eq } from "drizzle-orm";
import {
  buildSystemPrompt,
  buildToolDefinitions,
  zCreateTask,
  zCreateCapture,
  zCreateEvent,
  type ProjectSummary,
} from "@hyperpolymath/jarvis-core";

export const runtime = "nodejs";
export const maxDuration = 60;

const TOOL_VALIDATORS = {
  create_task: zCreateTask,
  create_capture: zCreateCapture,
  create_event: zCreateEvent,
} as const;

function sse(event: string, data: unknown): string {
  return `event: ${event}\ndata: ${JSON.stringify(data)}\n\n`;
}

export async function POST(req: NextRequest) {
  const startTime = Date.now();
  let firstTokenAt: number | null = null;

  // 1. Auth — re-derive userId at boundary (JARVIS-12)
  const supabase = await createClient();
  const { data: authData, error: authError } = await supabase.auth.getClaims();
  if (authError || !authData?.claims?.sub) return new Response("Unauthorized", { status: 401 });
  const userId = authData.claims.sub;

  // 2. Voice header (Phase 7 forward-compat)
  const voiceActive = req.headers.get("X-Voice-Active") === "true";

  // 3. Request body
  const body = (await req.json()) as {
    input: string;
    history: Array<{ role: "user" | "assistant"; content: string }>;
    parsedDates?: Array<{ text: string; start: string; end?: string; allDay?: boolean }>;
    slashCommand?: "task" | "capture" | "event" | "help" | null;
    linkedProjectIds?: string[];           // M5: client-validated project UUIDs from $ chips
    linkedHashtags?: string[];             // M6: client-extracted hashtag names from # chips
  };

  // 4. Load user context (projects + tz + default cal)
  const userProjects = await db.select({ id: projects.id, name: projects.name, icon: projects.icon })
    .from(projects)
    .where(eq(projects.userId, userId));
  const [userRow] = await db.select({
    timezone: users.timezone,
    defaultCalendarId: users.gcalDefaultCalendarId,
  }).from(users).where(eq(users.id, userId));

  const projectSummaries: ProjectSummary[] = userProjects.map((p) => ({ id: p.id, name: p.name, icon: p.icon }));
  const system = buildSystemPrompt({ projects: projectSummaries, voiceActive });
  const tools = buildToolDefinitions({ voiceActive });

  // 5. Build messages — slash-command forcing via tool_choice
  const toolChoice = body.slashCommand && body.slashCommand !== "help"
    ? { type: "tool" as const, name: `create_${body.slashCommand}` }
    : { type: "auto" as const };

  // Build user message — include pre-parsed dates as a hint block
  let userContent = body.input;
  if (body.parsedDates && body.parsedDates.length > 0) {
    userContent += `\n\n[Pre-parsed dates (use these ISO values, do not re-parse): ${JSON.stringify(body.parsedDates)}]`;
  }
  // M5+M6: surface client-resolved references so the model has validated IDs available.
  // The model still resolves names via buildProjectListContext, but with these IDs as a hint
  // it can short-circuit ambiguous-name lookups when chips were explicitly inserted.
  if ((body.linkedProjectIds?.length ?? 0) > 0 || (body.linkedHashtags?.length ?? 0) > 0) {
    const parts: string[] = [];
    if (body.linkedProjectIds && body.linkedProjectIds.length > 0) parts.push(`projects=${JSON.stringify(body.linkedProjectIds)}`);
    if (body.linkedHashtags && body.linkedHashtags.length > 0) parts.push(`hashtags=${JSON.stringify(body.linkedHashtags)}`);
    userContent += `\n\n[Linked references in this message (client-validated): ${parts.join(", ")}]`;
  }
  const messages: Array<{ role: "user" | "assistant"; content: string }> = [...body.history, { role: "user", content: userContent }];

  // 6. AbortController propagation
  const upstream = new AbortController();
  const onAbort = () => upstream.abort();
  req.signal.addEventListener("abort", onAbort, { once: true });

  // 7. Stream
  const encoder = new TextEncoder();
  const actionTypes: string[] = [];
  const executor = createServerExecutor();
  const ctx = { userId, userTimezone: userRow?.timezone ?? "America/New_York", defaultCalendarId: userRow?.defaultCalendarId ?? null };

  const stream = new ReadableStream({
    async start(controller) {
      const anth = getAnthropicClient();
      const anthStream = anth.messages.stream(
        {
          model: JARVIS_MODEL,
          max_tokens: 1024,
          system,
          tools,
          tool_choice: toolChoice,
          messages,
        },
        { signal: upstream.signal },
      );

      anthStream.on("contentBlock", async (block) => {
        if (firstTokenAt === null) firstTokenAt = Date.now() - startTime;
        if (block.type !== "tool_use") return;

        const validator = TOOL_VALIDATORS[block.name as keyof typeof TOOL_VALIDATORS];
        if (!validator) {
          controller.enqueue(encoder.encode(sse("error", { message: `Unknown tool: ${block.name}` })));
          return;
        }
        const parsed = validator.safeParse(block.input);
        if (!parsed.success) {
          controller.enqueue(encoder.encode(sse("error", { message: `Tool validation failed: ${parsed.error.message}` })));
          return;
        }

        actionTypes.push(block.name);
        let result;
        if (block.name === "create_task") result = await executor.createTask(parsed.data as any, ctx);
        else if (block.name === "create_capture") result = await executor.createCapture(parsed.data as any, ctx);
        else if (block.name === "create_event") result = await executor.createEvent(parsed.data as any, ctx);
        else return;

        controller.enqueue(encoder.encode(sse("action", { toolUseId: block.id, name: block.name, result })));
      });

      anthStream.on("text", (delta) => {
        if (firstTokenAt === null) firstTokenAt = Date.now() - startTime;
        controller.enqueue(encoder.encode(sse("text", { delta })));
      });

      try {
        const final = await anthStream.finalMessage();
        controller.enqueue(encoder.encode(sse("done", { usage: final.usage })));

        // Fire-and-forget telemetry
        void logJarvisEvent({
          userId,
          promptText: body.input,
          preParsedDates: body.parsedDates ?? null,
          slashCommandMode: body.slashCommand ?? null,
          voiceActive,
          actionTypes,
          usage: final.usage,
          latencyMs: Date.now() - startTime,
          firstTokenMs: firstTokenAt ?? undefined,
        });
      } catch (err: any) {
        if (err?.name !== "AbortError") {
          controller.enqueue(encoder.encode(sse("error", { message: String(err?.message ?? err) })));
          void logJarvisEvent({
            userId,
            promptText: body.input,
            voiceActive,
            actionTypes,
            latencyMs: Date.now() - startTime,
            firstTokenMs: firstTokenAt ?? undefined,
            error: String(err?.message ?? err),
          });
        }
      } finally {
        req.signal.removeEventListener("abort", onAbort);
        controller.close();
      }
    },
    cancel() {
      upstream.abort();
    },
  });

  return new Response(stream, {
    headers: {
      "Content-Type": "text/event-stream",
      "Cache-Control": "no-cache, no-transform",
      "Connection": "keep-alive",
      "X-Accel-Buffering": "no",
    },
  });
}
```

```typescript
import { z } from "zod";

import { captures, tasks, tasksProjects, capturesProjects } from "@/lib/db/schema";
import { createClient } from "@/lib/supabase/server";

const ConvertSchema = z.object({
  captureId: z.string().uuid(),
  title: z.string().min(1).max(500),
  priority: z.enum(["P∞", "P1", "P2", "P3"]).default("P3"),
  projectIds: z.array(z.string().uuid()).default([]),
});

export async function convertCaptureToTask(input: z.infer<typeof ConvertSchema>) {
  const supabase = await createClient();
  const { data: claims } = await supabase.auth.getClaims();
  if (!claims?.claims?.sub) return { ok: false as const, error: "Unauthorized" };
  const userId = claims.claims.sub;

  const parsed = ConvertSchema.safeParse(input);
  if (!parsed.success) return { ok: false as const, error: parsed.error.message };

  const [capture] = await db.select().from(captures)
    .where(and(eq(captures.id, parsed.data.captureId), eq(captures.userId, userId)));
  if (!capture) return { ok: false as const, error: "Capture not found" };

  const taskId = randomUUID();
  try {
    await db.transaction(async (tx) => {
      await tx.insert(tasks).values({
        id: taskId,
        userId,
        title: parsed.data.title,
        priority: parsed.data.priority,
        status: "not started",
      });
      if (parsed.data.projectIds.length > 0) {
        await tx.insert(tasksProjects).values(parsed.data.projectIds.map((pid) => ({ taskId, projectId: pid })));
      }
      await tx.delete(captures).where(eq(captures.id, parsed.data.captureId));
      // capturesProjects rows cascade-delete via FK
    });
    return { ok: true as const, taskId };
  } catch (err) {
    return { ok: false as const, error: String(err) };
  }
}
```

```typescript
  const reader = response.body!.getReader();
  const decoder = new TextDecoder();
  let buffer = "";
  while (true) {
    const { done, value } = await reader.read();
    if (done) break;
    buffer += decoder.decode(value);
    let i = buffer.indexOf("\n\n");
    while (i !== -1) {
      const chunk = buffer.slice(0, i);
      buffer = buffer.slice(i + 2);
      const eventName = chunk.match(/^event: (\w+)$/m)?.[1] ?? "";
      const data = JSON.parse(chunk.match(/^data: (.+)$/m)?.[1] ?? "null");
      events.push({ event: eventName, data });
      i = buffer.indexOf("\n\n");
    }
  }
  return events;
}
```
