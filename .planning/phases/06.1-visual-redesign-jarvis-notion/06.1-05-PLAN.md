`NavLink` sub-component:
```tsx
import { motion } from 'motion/react';
import { usePathname } from 'next/navigation';

interface NavLinkProps {
  href: string;
  label: string;
  isAgent?: boolean;  // when true and route active → cyan 4px dot
  count?: number;
}

function NavLink({ href, label, isAgent = false, count }: NavLinkProps) {
  const pathname = usePathname();
  const isActive = pathname === href || pathname.startsWith(`${href}/`);
  return (
    <a
      href={href}
      className={[
        'flex items-center gap-2 px-4 py-2 cursor-pointer-always',
        'font-mono text-xs uppercase tracking-[0.06em]',
        'border-l-2 border-l-transparent',
        isActive ? 'border-l-[var(--edge-hud)] text-[var(--ink)]' : 'text-[var(--ink-muted)] hover:text-[var(--ink)]',
        'transition-colors duration-100 ease-out',
      ].join(' ')}
      aria-current={isActive ? 'page' : undefined}
    >
      <span>{label}</span>
      {isActive && isAgent && (
        <motion.span
          initial={{ scale: 0 }}
          animate={{ scale: 1 }}
          transition={{ duration: 0.2, ease: [0.25, 1, 0.5, 1] }}
          className="inline-block w-1 h-1 rounded-full"
          style={{ backgroundColor: 'var(--hud-cyan)' }}
          aria-hidden="true"
        />
      )}
      {count != null && (
        <span className="ml-auto font-mono text-[11px] text-[var(--ink-muted)]">{count}</span>
      )}
    </a>
  );
}
```

Key interaction states:
- idle: text --ink-muted, border-l-transparent
- hover: text --ink (via Tailwind hover:)
- active: border-l-[var(--edge-hud)] (1px) + text --ink
- active + isAgent: + 4px cyan dot
- 100ms transition
