```tsx
'use client';

import { BarChart, Bar, LineChart, Line, XAxis, YAxis, CartesianGrid, ResponsiveContainer, Tooltip } from 'recharts';
import { useReducedMotion } from 'motion/react';
import { HudCornerCrops } from '@/components/shared/HudCornerCrops';

// Color literals — recharts SVG can't resolve CSS variables (UI-SPEC §3 + 06-04-SUMMARY pattern)
const HUD_CYAN = '#22d3ee';
const HUD_CYAN_DIM = '#0891b2';
const EDGE = '#d4cfc4';   // light mode grid hairline; for dark, conditionally pick #202938

interface ChartPanelProps {
  title: string;
  subtitle?: string;
  height: number;
  children: React.ReactNode;
}

function ChartPanel({ title, subtitle, height, children }: ChartPanelProps) {
  return (
    <div
      className="relative bg-[var(--surface)] rounded-sm p-6"
      style={{
        border: '1px solid var(--edge-hud)',
        boxShadow: '0 0 24px var(--hud-cyan-glow-soft)',
      }}
    >
      <HudCornerCrops size={8} breathing={false} className="absolute inset-0 pointer-events-none" />
      <header className="mb-4">
        <h2 className="font-mono text-[11px] uppercase tracking-[0.08em] text-[var(--ink-muted)]">{title}</h2>
        {subtitle && <p className="font-mono text-[11px] uppercase tracking-[0.08em] text-[var(--ink-muted)] opacity-60">{subtitle}</p>}
      </header>
      <div style={{ height }} className="relative">
        {children}
      </div>
    </div>
  );
}

export function InsightsCharts({ data }: { data: InsightsDataShape }) {
  const shouldReduce = useReducedMotion();
  return (
    <div className="grid grid-cols-1 md:grid-cols-2 gap-6">
      {/* Action distribution (full width on mobile, half on md+) */}
      <ChartPanel title="ACTION DISTRIBUTION" height={200}>
        <ResponsiveContainer width="100%" height="100%">
          <BarChart data={data.actionDistribution}>
            <CartesianGrid stroke={EDGE} strokeDasharray="2 2" vertical={false} />
            <XAxis
              dataKey="action"
              tick={{ fontSize: 11, fontFamily: 'var(--font-jetbrains-mono)', fill: 'var(--ink-muted)' }}
              axisLine={{ stroke: EDGE }}
              tickLine={false}
            />
            <YAxis hide />
            <Bar dataKey="count" fill={HUD_CYAN} isAnimationActive={!shouldReduce} animationDuration={800} animationEasing="ease-out" />
          </BarChart>
        </ResponsiveContainer>
      </ChartPanel>

      {/* Latency P50/P95 */}
      <ChartPanel title="LATENCY  P50/P95" height={200}>
        <ResponsiveContainer width="100%" height="100%">
          <LineChart data={data.latencyByDay}>
            <CartesianGrid stroke={EDGE} strokeDasharray="2 2" vertical={false} />
            <XAxis
              dataKey="day"
              tick={{ fontSize: 11, fontFamily: 'var(--font-jetbrains-mono)', fill: 'var(--ink-muted)' }}
              axisLine={{ stroke: EDGE }}
              tickLine={false}
            />
            <YAxis hide />
            <Line type="monotone" dataKey="p50" stroke={HUD_CYAN} strokeWidth={1.5} dot={false} connectNulls={true} isAnimationActive={!shouldReduce} animationDuration={800} animationEasing="ease-out" />
            <Line type="monotone" dataKey="p95" stroke={HUD_CYAN_DIM} strokeWidth={1.5} strokeDasharray="3 3" dot={false} connectNulls={true} isAnimationActive={!shouldReduce} animationDuration={800} animationEasing="ease-out" />
          </LineChart>
        </ResponsiveContainer>
      </ChartPanel>

      {/* Error rate (full width) */}
      <ChartPanel title="ERROR RATE" height={120} subtitle={data.errorPercent != null ? `${data.errorPercent.toFixed(1)}%` : '—'}>
        <ResponsiveContainer width="100%" height="100%">
          <LineChart data={data.errorByDay}>
            <Line type="monotone" dataKey="errorRate" stroke={HUD_CYAN_DIM} strokeWidth={1} dot={false} isAnimationActive={!shouldReduce} animationDuration={800} animationEasing="ease-out" />
          </LineChart>
        </ResponsiveContainer>
      </ChartPanel>
    </div>
  );
}
```

If existing `InsightsDataShape` differs, adapt the data field accesses but keep the visual contract.

  value: string;
