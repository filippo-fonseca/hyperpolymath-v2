    **Step 3: Create `apps/desktop/src/wake-word/worker.js`** — openWakeWord pipeline in plain ES module JS so the Worker can load it directly. The worker self-contains: init, frame buffering, mel inference, embedding inference, classifier inference, 2-consecutive-frame gating.

    Reference protocol (from Phase 12 RESEARCH §Pattern 6 and openWakeWord standard pipeline):
    ```javascript
    import * as ort from "onnxruntime-web";

    let melSession = null;
    let embSession = null;
    let cls Session = null;
    let threshold = 0.5;
    let consecutive = 0;
    let frameCount = 0;
    const SAMPLE_BUF = []; // PCM ring buffer awaiting mel inference (1280 samples = 80ms @ 16kHz)
    const EMB_BUF = []; // Embedding ring buffer (16 embeddings for classifier input)
    const FRAME_SAMPLES = 1280; // 80ms @ 16kHz

    self.onmessage = async (e) => {
      const msg = e.data;
      if (msg.type === "init") {
        threshold = msg.threshold ?? 0.5;
        melSession = await ort.InferenceSession.create("/src/wake-word/melspectrogram.onnx", {
          executionProviders: ["wasm"],
        });
        embSession = await ort.InferenceSession.create("/src/wake-word/embedding_model.onnx", {
          executionProviders: ["wasm"],
        });
        clsSession = await ort.InferenceSession.create("/src/wake-word/hey_jarvis_v0.1.onnx", {
          executionProviders: ["wasm"],
        });
        self.postMessage({ type: "ready" });
        return;
      }
      if (msg.type === "set-threshold") {
        threshold = msg.threshold;
        return;
      }
      if (msg.type === "frame") {
        const pcm = msg.pcm; // Float32Array
        for (let i = 0; i < pcm.length; i++) SAMPLE_BUF.push(pcm[i]);
        while (SAMPLE_BUF.length >= FRAME_SAMPLES) {
          const frame = SAMPLE_BUF.splice(0, FRAME_SAMPLES);
          await processFrame(new Float32Array(frame));
        }
      }
    };

    async function processFrame(frame) {
      try {
        const melInput = new ort.Tensor("float32", frame, [1, frame.length]);
        const melResult = await melSession.run({ [melSession.inputNames[0]]: melInput });
        const melOut = melResult[melSession.outputNames[0]];
        const embResult = await embSession.run({ [embSession.inputNames[0]]: melOut });
        const embOut = embResult[embSession.outputNames[0]];
        EMB_BUF.push(embOut.data);
        if (EMB_BUF.length > 16) EMB_BUF.shift();
        if (EMB_BUF.length < 16) return;
        const concat = new Float32Array(16 * embOut.data.length);
        for (let i = 0; i < 16; i++) concat.set(EMB_BUF[i], i * embOut.data.length);
        const clsInput = new ort.Tensor("float32", concat, [1, 16, embOut.data.length]);
        const clsResult = await clsSession.run({ [clsSession.inputNames[0]]: clsInput });
        const score = clsResult[clsSession.outputNames[0]].data[0];
        frameCount++;
        if (score > threshold) {
          consecutive++;
          if (consecutive >= 2) {
            self.postMessage({ type: "wake-fire", score, frameCount });
            consecutive = 0;
            EMB_BUF.length = 0;
          }
        } else {
          consecutive = 0;
        }
      } catch (err) {
        self.postMessage({ type: "error", message: String(err) });
      }
    }
    ```
    **CAVEAT**: the exact tensor shapes and input/output names depend on the specific ONNX models. If shapes mismatch at runtime, the Plan 05 e2e checkpoint will catch it — the executor should validate by spawning the worker in a small test harness or by reading the model metadata via `melSession.inputNames` and `melSession.outputNames` console log on first run.

    ```typescript
    import { listen, type UnlistenFn } from "@tauri-apps/api/event";

    import { startCaptureTurn } from "@/audio/capture";

    interface AudioChunkPayload { samples: number[]; sample_rate: number }

    let worker: Worker | null = null;
    let unlisten: UnlistenFn | null = null;
    let ready = false;
    let lastFireAt = 0;
    let debounceMs = 250;

    export async function startStandaloneWakeWord(opts: {
      threshold: number;
      debounceMs: number;
    }): Promise<void> {
      if (worker) return;
      debounceMs = opts.debounceMs;

      worker = new Worker(new URL("./worker.js", import.meta.url), { type: "module" });
      worker.onmessage = (e) => {
        const msg = e.data as { type: string; score?: number; message?: string };
        if (msg.type === "ready") {
          ready = true;
          // eslint-disable-next-line no-console
          console.log("[wake-word] worker ready");
        } else if (msg.type === "wake-fire") {
          const now = Date.now();
          if (now - lastFireAt < debounceMs) return;
          lastFireAt = now;
          // eslint-disable-next-line no-console
          console.log(`[wake-word] fire score=${msg.score?.toFixed(3)}`);
          void startCaptureTurn();
        } else if (msg.type === "error") {
          // eslint-disable-next-line no-console
          console.warn(`[wake-word] worker error: ${msg.message}`);
        }
      };
      worker.postMessage({ type: "init", threshold: opts.threshold });

      unlisten = await listen<AudioChunkPayload>("audio-chunk", (event) => {
        if (!ready || !worker) return;
        const pcm = new Float32Array(event.payload.samples);
        worker.postMessage({ type: "frame", pcm }, [pcm.buffer]);
      });

      // The cpal stream needs to be running continuously for wake-word inference.
      // startCaptureTurn() will start the stream; we must start it ourselves here.
      // The audio module currently only starts on capture-turn — extend it OR add a
      // dedicated "always-on" mode. For this plan: invoke start_capture directly here
      // and tolerate that startCaptureTurn's invoke is a no-op when stream is already running.
      const { invoke } = await import("@tauri-apps/api/core");
      await invoke("start_capture");
    }

    export function setWakeWordThreshold(threshold: number): void {
      worker?.postMessage({ type: "set-threshold", threshold });
    }

    export async function stopStandaloneWakeWord(): Promise<void> {
      if (unlisten) {
        unlisten();
        unlisten = null;
      }
      if (worker) {
        worker.terminate();
        worker = null;
        ready = false;
      }
      const { invoke } = await import("@tauri-apps/api/core");
      await invoke("stop_capture");
    }
    ```

    ```typescript
    let refcount = 0;

    export async function acquireStream(): Promise<void> {
      refcount++;
      if (refcount === 1) await invoke("start_capture");
    ```

    **Step 1: `apps/desktop/src/settings/store.ts`** — typed settings reader/writer over `@tauri-apps/plugin-store` + event emitter for live-apply:
    ```typescript
    import { Store } from "@tauri-apps/plugin-store";

    export interface DesktopSettings {
      wakeMode: "extender" | "standalone" | "both";
      vadSilenceMs: number;
      triggerDebounceMs: number;
      wakeWordScoreThreshold: number;
      transcribeEndpoint: string;
      verboseLog: boolean;
    }

    export const DEFAULT_SETTINGS: DesktopSettings = {
      wakeMode: "extender",
      vadSilenceMs: 1000,
      triggerDebounceMs: 250,
      wakeWordScoreThreshold: 0.5,
      transcribeEndpoint: "http://localhost:3000",
      verboseLog: false,
    };

    const STORE_FILE = "settings.json";
    const STORE_KEY = "desktop-settings";

    let cached: DesktopSettings | null = null;
    let storeInstance: Store | null = null;

    async function getStore(): Promise<Store> {
      if (!storeInstance) {
        storeInstance = await Store.load(STORE_FILE);
      }
      return storeInstance;
    }

    export async function loadSettings(): Promise<DesktopSettings> {
      if (cached) return cached;
      const store = await getStore();
      const raw = (await store.get<Partial<DesktopSettings>>(STORE_KEY)) ?? {};
      cached = { ...DEFAULT_SETTINGS, ...raw };
      return cached;
    }

    export async function saveSetting<K extends keyof DesktopSettings>(
      key: K,
      value: DesktopSettings[K],
    ): Promise<void> {
      const current = await loadSettings();
      const next = { ...current, [key]: value };
      cached = next;
      const store = await getStore();
      await store.set(STORE_KEY, next);
      await store.save();
      window.dispatchEvent(
        new CustomEvent("desktop-settings-change", { detail: { key, value, settings: next } }),
      );
    }

    export function subscribeToSettings(
      handler: (settings: DesktopSettings, changedKey: keyof DesktopSettings) => void,
    ): () => void {
      const listener = (e: Event) => {
        const detail = (e as CustomEvent).detail as {
          key: keyof DesktopSettings;
          settings: DesktopSettings;
        };
        handler(detail.settings, detail.key);
      };
      window.addEventListener("desktop-settings-change", listener);
      return () => window.removeEventListener("desktop-settings-change", listener);
    }
    ```

    ```rust
    pub fn setup_tray(app: &App) -> tauri::Result<()> {

        let quit_item = MenuItem::with_id(app, "quit", "Quit JARVIS Desktop", true, None::<&str>)?;

        let _tray = TrayIconBuilder::new()

            .tooltip("JARVIS Desktop")
            .menu(&menu)
            .show_menu_on_left_click(true)
            .on_menu_event(|app_handle, event| match event.id.as_ref() {
                "settings" => {

                }
                "quit" => {
                    app_handle.exit(0);
                }
                _ => {}
            })
            .on_tray_icon_event(|_tray, _event: TrayIconEvent| {})
            .build(app)?;
    ```

    - Subscribes to settings changes; on `wakeMode` change starts/stops the relevant listener; on `wakeWordScoreThreshold` change calls `setWakeWordThreshold`

    ```typescript
    import { listen } from "@tauri-apps/api/event";

    import { startPhysicalExtenderListener, stopPhysicalExtenderListener } from "@/physical-extender/sse-client";
    import { startStandaloneWakeWord, stopStandaloneWakeWord, setWakeWordThreshold } from "@/wake-word/standalone";
    import { loadSettings, subscribeToSettings, type DesktopSettings } from "@/settings/store";
    import { openSettingsWindow } from "@/settings/window";

    async function applyMode(settings: DesktopSettings): Promise<void> {
      const wantsExtender = settings.wakeMode === "extender" || settings.wakeMode === "both";
      const wantsStandalone = settings.wakeMode === "standalone" || settings.wakeMode === "both";
      if (wantsExtender) startPhysicalExtenderListener();
      else stopPhysicalExtenderListener();
      if (wantsStandalone) {
        await startStandaloneWakeWord({
          threshold: settings.wakeWordScoreThreshold,
          debounceMs: settings.triggerDebounceMs,
        });
      } else {
        await stopStandaloneWakeWord();
      }
    }

    async function boot(): Promise<void> {
      const settings = await loadSettings();
      await applyMode(settings);

      subscribeToSettings(async (next, changedKey) => {
        if (changedKey === "wakeMode" || changedKey === "triggerDebounceMs") {
          await applyMode(next);
        } else if (changedKey === "wakeWordScoreThreshold") {
          setWakeWordThreshold(next.wakeWordScoreThreshold);
        }
        // vadSilenceMs is read by the next capture turn via loadSettings — no restart needed.
      });

      await listen("tray-open-settings", () => {
        void openSettingsWindow();
      });

      // eslint-disable-next-line no-console
      console.log(`[boot] JARVIS Desktop ready — mode=${settings.wakeMode}`);
    }

    boot().catch((err) => {
      // eslint-disable-next-line no-console
      console.error("[boot]", err);
    });
    ```
