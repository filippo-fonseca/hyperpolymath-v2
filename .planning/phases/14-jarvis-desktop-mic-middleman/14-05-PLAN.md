          return { skip: true };

        // Confirm the workspace exists (defensive — if Plan 14-01 hasn't shipped, skip gracefully).
        try {
          readdirSync(resolve(DESKTOP_DIR, "src-tauri"));
        } catch {
          warn("desktop", "apps/desktop/src-tauri not found — desktop not scaffolded yet");
          return { skip: true };
        }
      },
      start: () =>
        spawn("pnpm", ["--filter", "desktop", "tauri", "dev"], {
          cwd: REPO_ROOT,
          stdio: ["ignore", "pipe", "pipe"],
        }),
      keepAlive: true,
      ready: (proc) =>
        waitForLog(
          proc,
          /Running .*http:\/\/localhost|App is ready|JARVIS Desktop ready/i,
          180_000,
        ),
    },
