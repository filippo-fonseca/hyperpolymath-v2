import { AudioQueue } from "./audio-queue";
import { DEFAULT_VOICE_ID } from "./constants";
import { collectStage } from "./voice-stage-collector";
import type { VoiceSettings } from "./types";

export interface TurnPlaybackParams {
  voiceId: string;
  ttsProvider: VoiceSettings["ttsProvider"];
  audioContext: AudioContext;
  onStart: () => void;   // fired on first audio-out start of the turn
  onEnd: () => void;     // fired when ALL queued audio has drained + no in-flight fetches remain
}

/**
 * Per-turn playback orchestrator.
 *
 * Lifecycle:
 *   1. Created on first playSentence(text, 0, params) by useTtsPlayer.
 *   2. Each playSentence schedules a fetch + appends chunks to a
 *      per-sequence pending-chunk buffer.
 *   3. enqueueGate (Map<seq, Promise<void>>) serializes AudioQueue.enqueue
 *      across sequences in seq-order so audio plays in-order even when
 *      fetches resolve out of order.
 *   4. endOfTurn() marks the controller closed; once all in-flight fetches
 *      drain AND AudioQueue.onAllEnded fires, onEnd is called and the
 *      controller can be discarded.
 *   5. stop() aborts every in-flight fetch + AudioQueue.stopAll() + fires
 *      onEnd if not already fired.
 *
 * Fallback policy (D-04):
 *   - If first sentence (seq === 0) fails (502 / network / abort-other):
 *     switch fallbackVoice → 'browser' and replay sentence 0 via
 *     SpeechSynthesis. ALL subsequent playSentence calls go through
 *     SpeechSynthesis until endOfTurn. No mid-turn voice swap.
 *   - If sentence ≥ 1 fails after sentence 0 already played via ElevenLabs:
 *     silent drop + console.warn. Continue with subsequent sentences via
 *     ElevenLabs.
 *
 * Telemetry (D-06):
 *   - tts_first_byte_at: collectStage fires ONCE per turn, on the FIRST
 *     /api/jarvis/tts response (sentence 0's first fetch). Subsequent
 *     fetches do not re-fire.
 *   - audio_first_play_at: AudioQueue.firstPlayCaptured already enforces
 *     once-per-AudioQueue-lifetime; the controller creates one AudioQueue
 *     per turn, so this is correctly first-sentence-of-turn semantics.
 */
export class TurnPlaybackController {
  private audioQueue: AudioQueue;
  private inFlight: Map<number, AbortController> = new Map();
  private enqueueGate: Map<number, Promise<void>> = new Map();
  private fallbackVoice: "elevenlabs" | "browser" = "elevenlabs";
  private firstByteCaptured = false;
  private endOfTurnCalled = false;
  private stopped = false;
  private params: TurnPlaybackParams;
  private firstAudioStarted = false;
  private lastSeq = -1;

  constructor(params: TurnPlaybackParams) {
    this.params = params;
    this.audioQueue = new AudioQueue(params.audioContext);
    this.audioQueue.onAllEnded(() => this.maybeFireOnEnd());
  }

  /**
   * Dispatch a sentence for playback at the given sequence index.
   * Sequence index MUST be monotonic per turn starting at 0.
   * Returns immediately; audio is played asynchronously.
   */
  async playSentence(text: string, seq: number): Promise<void> {
    if (this.stopped) return;
    this.lastSeq = Math.max(this.lastSeq, seq);

    if (this.fallbackVoice === "browser") {
      this.speakViaSpeechSynthesis(text);
      return;
    }

    const ctrl = new AbortController();
    this.inFlight.set(seq, ctrl);
    const timeout = setTimeout(() => ctrl.abort(), 8000);  // per-sentence 8s

    // Initialize the in-order enqueue gate for this seq.
    const prevGate = seq === 0
      ? Promise.resolve()
      : this.enqueueGate.get(seq - 1) ?? Promise.resolve();

    let resolveThisGate!: () => void;
    this.enqueueGate.set(seq, new Promise<void>((r) => { resolveThisGate = r; }));

    try {
      const res = await fetch("/api/jarvis/tts", {
        method: "POST",
        headers: { "Content-Type": "application/json" },
        body: JSON.stringify({
          text,
          voiceId: this.params.voiceId || DEFAULT_VOICE_ID,
        }),
        signal: ctrl.signal,
      });
      clearTimeout(timeout);

      // Phase 9 / TEL-01 (D-06): first-sentence one-shot — fire once per turn.
      if (!this.firstByteCaptured) {
        this.firstByteCaptured = true;
        collectStage("tts_first_byte_at", new Date());
      }

      if (res.status === 502 || !res.ok || !res.body) {
        throw new Error(`TTS upstream ${res.status}`);
      }

      const reader = res.body.getReader();
      // Wait for our predecessor's enqueue gate before starting to enqueue,
      // so out-of-order fetch resolution still produces in-order audio.
      await prevGate;

      while (true) {
        const { done, value } = await reader.read();
        if (done) break;
        if (!value || value.byteLength === 0) continue;
        // value is a Uint8Array; convert to ArrayBuffer for AudioQueue.enqueue.
        const buf = value.buffer.slice(
          value.byteOffset,
          value.byteOffset + value.byteLength,
        ) as ArrayBuffer;
        await this.audioQueue.enqueue(buf);
        if (!this.firstAudioStarted) {
          this.firstAudioStarted = true;
          this.params.onStart();
        }
      }
    } catch (err) {
      clearTimeout(timeout);
      const e = err as { name?: string; message?: string };
      if (e?.name === "AbortError") {
        // stop() was called or per-sentence timeout fired. If stop() called,
        // onEnd already cycled; if timeout, fall through to fallback logic.
        if (this.stopped) {
          resolveThisGate();
          this.inFlight.delete(seq);
          return;
        }
      }
      // Fallback policy (D-04):
      if (seq === 0 && !this.firstAudioStarted) {
        // First sentence failed BEFORE any audio played → switch the whole
        // turn to SpeechSynthesis. Replay this sentence via browser.
        console.warn("[tts] sentence 0 failed; falling back to SpeechSynthesis for turn", err);
        this.fallbackVoice = "browser";
        this.speakViaSpeechSynthesis(text);
      } else {
        // Sentence ≥ 1 OR sentence 0 after audio already started: silent
        // drop. Subsequent sentences continue via ElevenLabs.
        console.warn(`[tts] sentence ${seq} failed silently`, err);
      }
    } finally {
      resolveThisGate();
      this.inFlight.delete(seq);
      // If endOfTurn was called and this was the last in-flight, check fire.
      this.maybeFireOnEnd();
    }
  }

  /**
   * Signal that no more sentences will be dispatched. The controller fires
   * onEnd once all in-flight fetches drain AND AudioQueue empties.
   */
  endOfTurn(): void {
    this.endOfTurnCalled = true;
    this.maybeFireOnEnd();
  }

  /**
   * Stop everything immediately: abort every in-flight fetch, flush
   * AudioQueue, cancel any SpeechSynthesis. Fires onEnd (idempotent).
   */
  stop(): void {
    if (this.stopped) return;
    this.stopped = true;
    for (const ctrl of this.inFlight.values()) {
      try { ctrl.abort(); } catch { /* non-fatal */ }
    }
    this.inFlight.clear();
    this.audioQueue.stopAll();
    if (typeof window !== "undefined" && window.speechSynthesis) {
      window.speechSynthesis.cancel();
    }
    if (!this.endOfTurnCalled) {
      this.endOfTurnCalled = true;
      this.params.onEnd();
    }
  }

  private maybeFireOnEnd(): void {
    if (this.stopped) return;
    if (!this.endOfTurnCalled) return;
    if (this.inFlight.size > 0) return;
    // AudioQueue empties via onAllEnded; if it's already empty we still call.
    this.params.onEnd();
  }

  private speakViaSpeechSynthesis(text: string): void {
    if (typeof window === "undefined" || !window.speechSynthesis) {
      this.maybeFireOnEnd();
      return;
    }
    const u = new SpeechSynthesisUtterance(text);
    const voices = window.speechSynthesis.getVoices();
    const gb = voices.find((v) => v.lang === "en-GB");
    if (gb) u.voice = gb;
    u.onstart = () => {
      if (!this.firstAudioStarted) {
        this.firstAudioStarted = true;
        this.params.onStart();
      }
    };
    u.onend = () => this.maybeFireOnEnd();
    window.speechSynthesis.speak(u);
  }
}

export interface PlaySentenceParams {
  voiceId: string;
  ttsProvider: VoiceSettings["ttsProvider"];
  audioContext: AudioContext;
  onStart: () => void;
  onEnd: () => void;
}

export function useTtsPlayer() {
  const controllerRef = useRef<TurnPlaybackController | null>(null);

  const playSentence = useCallback(
    (text: string, seq: number, params: PlaySentenceParams): void => {
      // Silence / browser-direct branches fast-path (no controller needed for the
      // off branch — keeps FSM cycle correct).
      if (params.ttsProvider === "off" || !text.trim()) {
        if (seq === 0) params.onEnd();
        return;
      }
      if (params.ttsProvider === "browser") {
        // Direct SpeechSynthesis without ElevenLabs at all.
        // (Mirror the existing single-shot pattern — okay for browser provider.)
        if (!controllerRef.current) {
          controllerRef.current = new TurnPlaybackController({
            ...params,
            // Force fallback-voice path immediately.
          });
          // Set fallback via a public setter or constructor flag.
        }
        // Implementation: rely on the controller's speakViaSpeechSynthesis
        // path via a "forceFallback: true" param OR simply switch fallbackVoice
        // on the controller before first playSentence.
        controllerRef.current.playSentence(text, seq);
        return;
      }

      if (!controllerRef.current) {
        controllerRef.current = new TurnPlaybackController(params);
      }
      void controllerRef.current.playSentence(text, seq);
    },
    [],
  );

  const endOfTurn = useCallback((): void => {
    controllerRef.current?.endOfTurn();
    // Schedule controller disposal once it has drained — the controller's
    // onEnd callback (passed via params) is the right moment, NOT here.
  }, []);

  const stop = useCallback((): void => {
    controllerRef.current?.stop();
    controllerRef.current = null;
  }, []);

  return { playSentence, endOfTurn, stop };
}
