```typescript
// apps/web/lib/voice/sentence-splitter.ts (NEW)

/** Result of splitting one delta-tick of accumulated text. */
export interface SplitResult {
  /** Completed sentences (each retains its terminator: '. ', '! ', '? ', or '\n\n'). */
  sentences: string[];
  /** Unfinished tail to carry forward for the next call. */
  remainder: string;
}

/**
 * Pure sentence splitter for streaming Anthropic text deltas.
 *
 * Splits on the literal regex from REQUIREMENTS / D-02:
 *   - `. ` (period + space)
 *   - `! ` (exclamation + space)
 *   - `? ` (question + space)
 *   - `\n\n` (double newline)
 *
 * Usage pattern (caller owns the buffer):
 *   let buffer = "";
 *   // per delta:
 *   const { sentences, remainder } = splitDeltas(buffer, delta);
 *   buffer = remainder;
 *   for (const s of sentences) playSentence(s, seq++);
 *   // on stream close:
 *   if (buffer.trim()) playSentence(buffer, seq++);  // final flush
 *
 * @param prevBuffer The unfinished tail from prior calls (start with "").
 * @param newDelta   The latest text-delta arrival from the SSE stream.
 * @returns SplitResult with completed sentences + unfinished tail.
 */
export function splitDeltas(prevBuffer: string, newDelta: string): SplitResult;
```

    3. Implement using a single regex with `matchAll` to find all terminator positions, OR an iterative `String.prototype.indexOf` scan. EITHER is acceptable per D-02 "Claude's Discretion". Recommended idiom (regex-based, clearest):

       ```typescript
       export interface SplitResult {
         sentences: string[];
         remainder: string;
       }

       // Match a sentence terminator: `. `, `! `, `? `, or `\n\n` (one or more).
       // Per D-02 / REQUIREMENTS literal — NO library, NO minimum-length gate.
       const TERMINATOR = /([.!?] |\n\n+)/g;

       export function splitDeltas(prevBuffer: string, newDelta: string): SplitResult {
         if (!newDelta) {
           return { sentences: [], remainder: prevBuffer };
         }
         const combined = prevBuffer + newDelta;
         const sentences: string[] = [];
         let lastEnd = 0;
         for (const m of combined.matchAll(TERMINATOR)) {
           // m.index is the start of the terminator; end = m.index + m[0].length
           const end = (m.index ?? 0) + m[0].length;
           sentences.push(combined.slice(lastEnd, end));
           lastEnd = end;
         }
         const remainder = combined.slice(lastEnd);
         return { sentences, remainder };
       }
       ```
