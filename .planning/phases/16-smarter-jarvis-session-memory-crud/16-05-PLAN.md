    ```typescript
    type ContentBlock =
      | { type: "text"; text: string }
      | { type: "tool_use"; id: string; name: string; input: Record<string, unknown> }
      | { type: "tool_result"; tool_use_id: string; content: string };

    function reconstructToolInput(action: ScrollbackAction): Record<string, unknown> {
      // Reconstruct a minimal but valid input from the receipt. The model primarily
      // needs the id and a label for reference; perfect fidelity is not required.
      const r = action.result?.receipt ?? {};
      switch (action.name) {
        case "create_task":
        case "update_task":
          return { id: action.result?.id ?? r.id, title: r.title, status: r.status, priority: r.priority, due: r.dueDate };
        case "delete_task":
          return { id: r.id ?? action.result?.id };
        case "create_capture":
        case "update_capture":
          return { id: action.result?.id ?? r.id, content: r.content };
        case "delete_capture":
          return { id: r.id ?? action.result?.id };
        case "create_event":
        case "update_event":
          return { id: action.result?.id ?? r.id, calendar_id: r.calendar_id, title: r.title, start: r.start, end: r.end };
        case "delete_event":
          return { id: r.id ?? action.result?.id, calendar_id: r.calendar_id };
        case "find_tasks":
        case "find_captures":
        case "find_events":
          return { query: r.query }; // best effort
        case "ask_clarification":
          return { question: r.question };
        case "remember_fact":
          return { fact: r.fact, type: r.type };
        default:
          return {};
      }
    }

    function buildHistory(scrollback: ScrollbackTurn[]): Array<{ role: "user" | "assistant"; content: string | ContentBlock[] }> {
      const recent = scrollback.slice(-HISTORY_TURN_LIMIT);
      const out: Array<{ role: "user" | "assistant"; content: string | ContentBlock[] }> = [];
      for (const turn of recent) {
        if (turn.role === "user") {
          out.push({ role: "user", content: turn.text ?? "" });
          continue;
        }
        // Assistant turn — gather done actions
        const doneActions = (turn.actions ?? []).filter(a => a.status === "done");
        if (doneActions.length === 0) {
          // Plain prose assistant turn (no tool calls)
          if (turn.textDelta) out.push({ role: "assistant", content: turn.textDelta });
          continue;
        }
        // Mixed: text + tool_use blocks
        const assistantBlocks: ContentBlock[] = [];
        if (turn.textDelta) assistantBlocks.push({ type: "text", text: turn.textDelta });
        for (const a of doneActions) {
          assistantBlocks.push({ type: "tool_use", id: a.toolUseId, name: a.name, input: reconstructToolInput(a) });
        }
        out.push({ role: "assistant", content: assistantBlocks });
        // Anthropic REQUIRES that the immediately following turn carry tool_result blocks for
        // every tool_use id in the assistant turn — Pitfall 1.
        const toolResultBlocks: ContentBlock[] = doneActions.map(a => ({
          type: "tool_result",
          tool_use_id: a.toolUseId,
          content: JSON.stringify(a.result ?? { ok: false, error: "no result persisted" }),
        }));
        out.push({ role: "user", content: toolResultBlocks });
      }
      return out;
    }
    ```
