<objective>
Final wiring — predictive cache-warmer per D-03. Three UX entry signals (app open / input focus / mic arm), debounced 30s per trigger, age-gated server-side to 50 minutes. Net cost ~$0.01-0.03/day vs heartbeat warmer's $0.90/day. The warmer fires a 1-token no-op against the SAME tools + system shape as the real route (NO snapshot — warming tiers 1+2 only) to keep the 1h cache window from cooling between sessions.

Purpose: keep tier 1+2 cache warm across the day without a polling heartbeat. The user perceives this as "JARVIS responds instantly even hours into the day."
Output: 1 new endpoint + 1 new client component + 3 small wire-ups + 2 test files.

    /**
     * POST /api/jarvis/warm — predictive cache warmer (Phase 11 / CACHE-04, D-03).
     *
     * Fires a 1-token no-op Anthropic call with the same tools + frozen-system
     * shape as the real /api/jarvis route. NO user-state snapshot — warming
     * tier 1 + tier 2 (the 1h TTL tiers) only. Body is empty; user message
     * is the literal string "warm".
     *
     * Triggers: app open / JARVIS input focus / mic arm — all dispatched by
     * JarvisWarmer client component. Server-side age-gate: skip if the cache
     * was warmed within the last 50min (well inside 1h TTL with margin).
     *
     * Net cost (single user): ~$0.01-0.03/day with 30s client debounce +
     * 50min server age-gate. Compare to a heartbeat warmer at ~$0.90/day.
     */

    import type { NextRequest } from "next/server";
    import { eq } from "drizzle-orm";
    import { db } from "@/lib/db";
    import { projects, users } from "@/lib/db/schema";
    import { getJarvisFactsForUser } from "@/lib/db/queries/jarvis-facts";
    import {
      getAnthropicClient,
      JARVIS_MODEL,
    } from "@/lib/jarvis/anthropic-client";
    import { createClient } from "@/lib/supabase/server";
    import {
      buildSystemPrompt,
      buildToolDefinitions,
      type ProjectSummary,
    } from "@hyperpolymath/jarvis-core";
    import { getLastWarmAt, setLastWarmAt } from "@/lib/jarvis/state-snapshot-cache";

    export const runtime = "nodejs";

    /** D-03 age-gate threshold: 50 minutes (well inside the 1h TTL window). */
    const AGE_GATE_MS = 50 * 60 * 1000;

    export async function POST(_req: NextRequest) {
      // 1. Auth — same pattern as /api/jarvis
      const supabase = await createClient();
      const claimsResult = await supabase.auth.getClaims();
      if (claimsResult.error || !claimsResult.data?.claims?.sub) {
        return new Response("Unauthorized", { status: 401 });
      }
      const userId = claimsResult.data.claims.sub;

      // 2. Age-gate per D-03 — skip if recently warmed.
      const lastWarmAt = getLastWarmAt(userId);
      const now = Date.now();
      if (lastWarmAt !== null && now - lastWarmAt < AGE_GATE_MS) {
        return new Response(null, { status: 204 });
      }

      // 3. Build tools + system identical to /api/jarvis. NO snapshot block
      //    appended — warming tier 1 + tier 2 (1h TTL) only.
      const [userProjects, _userRows, userFacts] = await Promise.all([
        db
          .select({ id: projects.id, name: projects.name, icon: projects.icon })
          .from(projects)
          .where(eq(projects.userId, userId)),
        db.select({ timezone: users.timezone }).from(users).where(eq(users.id, userId)).limit(1),
        getJarvisFactsForUser(userId),
      ]);
      const projectSummaries: ProjectSummary[] = userProjects.map((p) => ({
        id: p.id, name: p.name, icon: p.icon,
      }));
      const system = buildSystemPrompt({
        projects: projectSummaries,
        facts: userFacts as import("@hyperpolymath/jarvis-core").JarvisFact[],
        voiceActive: false, // Warm always non-voice; the voice variant is a sibling cache key the user will hit on first real voice turn anyway.
      });
      const tools = buildToolDefinitions({ voiceActive: false });

      // 4. Fire 1-token no-op. Use messages.create (not stream) — no streaming
      //    needed for a single-token call. tool_choice: none so the model
      //    doesn't try to actually emit a tool_use block.
      try {
        const anth = getAnthropicClient();
        const result = await anth.messages.create({
          model: JARVIS_MODEL,
          max_tokens: 1,
          system: system as unknown as never,
          tools: tools as unknown as never,
          tool_choice: { type: "none" } as unknown as never,
          messages: [{ role: "user", content: "warm" } as unknown as never],
        });

        // Record warm timestamp BEFORE returning so race conditions favor
        // skip-next over double-warm.
        setLastWarmAt(userId, now);

        const usage = (result as { usage?: { cache_read_input_tokens?: number; cache_creation_input_tokens?: number } }).usage ?? {};
        return Response.json({
          cacheRead: usage.cache_read_input_tokens ?? 0,
          cacheCreate: usage.cache_creation_input_tokens ?? 0,
        });
      } catch (err) {
        console.error("[jarvis] warm endpoint failed", err);
        return new Response("Warm failed", { status: 500 });
      }
    }
    ```

    "use client";

    /**
     * Phase 11 / CACHE-04 (D-03) — predictive cache warmer.
     *
     * Mounted in app/(app)/layout.tsx — covers every authenticated route once.
     *
     * Triggers a fire-and-forget POST /api/jarvis/warm on three UX signals:
     *   - Mount: app open (once per component mount)
     *   - window CustomEvent("jarvis-input-focus"): JarvisInput onFocus
     *   - window CustomEvent("mic-arm"):           mic FSM idle → listening
     *
     * Per-trigger client-side debounce: 30s. Server-side age-gate (50min)
     * lives in /api/jarvis/warm and provides defense-in-depth — if a future
     * change accidentally drops the debounce, the server still skips.
     */

    import { useEffect, useRef } from "react";

    type WarmTrigger = "mount" | "input-focus" | "mic-arm";
    const DEBOUNCE_MS = 30 * 1000;

    export function JarvisWarmer(): null {
      // Per-trigger last-fire timestamps. Refs (not state) — debounce decisions
      // never need to trigger a re-render. Map preserves cross-trigger
      // isolation per D-03.
      const lastFiredRef = useRef<Map<WarmTrigger, number>>(new Map());

      useEffect(() => {
        const fire = (trigger: WarmTrigger): void => {
          const now = Date.now();
          const last = lastFiredRef.current.get(trigger) ?? 0;
          if (now - last < DEBOUNCE_MS) return;
          lastFiredRef.current.set(trigger, now);
          // Fire-and-forget. The endpoint enforces the server-side age-gate
          // and returns 204 if recently warmed — we don't act on the response.
          fetch("/api/jarvis/warm", { method: "POST" }).catch(() => {
            // Network failures must NEVER surface to the user. Telemetry-only.
          });
        };

        // Trigger 1: mount.
        fire("mount");

        // Trigger 2: input-focus.
        const onInputFocus = (): void => fire("input-focus");
        window.addEventListener("jarvis-input-focus", onInputFocus);

        // Trigger 3: mic-arm.
        const onMicArm = (): void => fire("mic-arm");
        window.addEventListener("mic-arm", onMicArm);

        return () => {
          window.removeEventListener("jarvis-input-focus", onInputFocus);
          window.removeEventListener("mic-arm", onMicArm);
        };
      }, []);

      return null;
    }
    ```
