**jsdom visibilityState mocking (M4 fix):**

`Object.defineProperty(document, "visibilityState", ...)` must include `writable: true` (or `configurable: true` with a fresh descriptor each test) to avoid the "Cannot redefine property" error after the second test. Cleanest pattern: define ONCE in `beforeEach` with `writable: true`, then assign `.value` directly in each test:

```typescript
beforeEach(() => {
  Object.defineProperty(document, "visibilityState", {
    value: "visible",
    writable: true,
    configurable: true,
  });
});

it("...", () => {
  (document as { visibilityState: string }).visibilityState = "hidden";
  // ... assertion
  (document as { visibilityState: string }).visibilityState = "visible";
  document.dispatchEvent(new Event("visibilitychange"));
});
```

**Render `<QueryProvider>` directly in visibility-recovery test (M4 fix):**

Per checker M4, render the production `<QueryProvider>` rather than replicating its listener — exercises the actual code path that ships, not a parallel mirror that can drift. The `QueryProvider` creates its own internal QueryClient via `useState(() => makeQueryClient())`, so to spy on `invalidateQueries` we wrap it with a shim that exposes the client:

```typescript
import { render } from "@testing-library/react";
import { QueryClient, QueryClientProvider, useQueryClient } from "@tanstack/react-query";

// Strategy: render <QueryProvider> (production code) AND mount a child that
// captures its QueryClient via useQueryClient(). Spy invalidateQueries on the
// captured client. The visibilitychange listener inside QueryProvider operates
// on the same client, so the spy sees production invocations.
function CaptureClient({ onClient }: { onClient: (qc: QueryClient) => void }) {
  const qc = useQueryClient();
  React.useEffect(() => { onClient(qc); }, [qc, onClient]);
  return null;
}
```

    Step 1 — Create `apps/web/tests/realtime-echo-dedupe-integration.test.ts`. This test exercises the actual reducer + a simulated Realtime event flow, all in-memory (no DB needed):
    ```typescript
    import { describe, it, expect } from "vitest";
    import { optimisticReducer } from "@/lib/realtime/optimistic-reducer";

    type CaptureRow = {
      id: string;
      content: string;
      createdAt: Date;
      updatedAt: Date;
    };

    describe("Optimistic → Realtime echo dedupe (RT-05 end-to-end)", () => {
      it("client UUID flows through and Realtime echo is a no-op", () => {
        // 1. Initial state: empty feed
        let state: CaptureRow[] = [];

        // 2. Client generates UUID and dispatches optimistic insert
        const clientUuid = crypto.randomUUID();
        const optimisticRow: CaptureRow = {
          id: clientUuid,
          content: "live test",
          createdAt: new Date(),
          updatedAt: new Date(),
        };
        state = optimisticReducer<CaptureRow>(state, { type: "insert", row: optimisticRow });
        expect(state).toHaveLength(1);
        expect(state[0].id).toBe(clientUuid);

        // 3. Simulate Server Action persisting with the same UUID — the server
        //    respects the caller-supplied id (per Plan 03-02 Task 1 + 03-03 Task 1).
        const persistedRow: CaptureRow = {
          id: clientUuid,
          content: "live test",
          createdAt: optimisticRow.createdAt,
          updatedAt: new Date(),
        };

        // 4. Simulate Realtime echo: TanStack Query refetches and returns
        //    [persistedRow] as the canonical state. The reducer is now applied
        //    fresh to that canonical state, with the optimistic action still
        //    pending (this is the React 19 useOptimistic semantics):
        const canonical = [persistedRow];
        const merged = optimisticReducer<CaptureRow>(canonical, { type: "insert", row: optimisticRow });

        // 5. Assert: the echo is a no-op — no duplicate row.
        expect(merged).toHaveLength(1);
        expect(merged[0].id).toBe(clientUuid);
      });

      it("update echo with same id is idempotent", () => {
        const id = crypto.randomUUID();
        let state: CaptureRow[] = [{ id, content: "old", createdAt: new Date(), updatedAt: new Date() }];

        // Optimistic update
        state = optimisticReducer<CaptureRow>(state, { type: "update", id, patch: { content: "new" } });
        expect(state[0].content).toBe("new");

        // Realtime echo: server returns the canonical updated row; reducer reapplies
        const canonical = [{ id, content: "new", createdAt: state[0].createdAt, updatedAt: new Date() }];
        const merged = optimisticReducer<CaptureRow>(canonical, { type: "update", id, patch: { content: "new" } });
        expect(merged).toEqual(canonical);
      });

      it("delete echo with same id is idempotent (deleted stays deleted)", () => {
        const id = crypto.randomUUID();
        let state: CaptureRow[] = [{ id, content: "x", createdAt: new Date(), updatedAt: new Date() }];
        state = optimisticReducer<CaptureRow>(state, { type: "delete", id });
        expect(state).toEqual([]);

        // Realtime DELETE event: canonical refetch returns []
        const canonical: CaptureRow[] = [];
        const merged = optimisticReducer<CaptureRow>(canonical, { type: "delete", id });
        expect(merged).toEqual([]);
      });
    });
    ```
