The canonical optimistic reducer for a list-of-rows surface:

```typescript
import { useOptimistic } from "react";

type OptimisticAction<T extends { id: string }> =
  | { type: "insert"; row: T }
  | { type: "update"; id: string; patch: Partial<T> }
  | { type: "delete"; id: string }
  | { type: "reorder"; ids: string[] }; // optional, for ordered lists

function optimisticReducer<T extends { id: string }>(state: T[], action: OptimisticAction<T>): T[] {
  switch (action.type) {
    case "insert": {
      // Echo dedupe: if a row with the same id already exists, ignore (RT-05).
      // This is the linchpin of optimistic + Realtime echo correctness.
      if (state.some((r) => r.id === action.row.id)) return state;
      return [action.row, ...state];
    }
    case "update":
      return state.map((r) => (r.id === action.id ? { ...r, ...action.patch } : r));
    case "delete":
      return state.filter((r) => r.id !== action.id);
    case "reorder": {
      const map = new Map(state.map((r) => [r.id, r]));
      return action.ids.map((id) => map.get(id)).filter((r): r is T => Boolean(r));
    }
    default:
      return state;
  }
}

// Per-domain hook usage:
const [optimisticTasks, addOptimistic] = useOptimistic(tasks, optimisticReducer<Task>);

// Mutation site:
const newId = crypto.randomUUID();
const optimisticRow: Task = { id: newId, ...input, createdAt: new Date(), updatedAt: new Date() };
addOptimistic({ type: "insert", row: optimisticRow });
const result = await createTask({ id: newId, ...input });
if (!result.success) {
  toast.error(result.error);
  // useOptimistic auto-reverts when the transition completes without committing real state.
  return;
}
// Real state arrives via Realtime echo → invalidateQueries → refetch.
// The refetched row already has the same id, so the deduped optimistic insert
// stops mattering once the canonical row lands.
```

```typescript
// apps/web/app/actions/tasks.ts (Task 1 adds)
"use server";
import { createClient } from "@/lib/supabase/server";
import { db } from "@/lib/db";
import { tasks } from "@/lib/db/schema";
import { eq } from "drizzle-orm";

export async function getTasksForCurrentUser(): Promise<Task[]> {
  const supabase = await createClient();
  const { data: { claims } } = await supabase.auth.getClaims();
  if (!claims) throw new Error("Unauthorized");
  return db.select().from(tasks).where(eq(tasks.userId, claims.sub));
}
```

**Canonical detail-page pattern (B1 fix — single source of truth):**

For a "show one record" page (project detail, task detail panel, capture detail panel), DO NOT use a per-id query key like `["project", projectId]`. Per-id keys are not invalidated by Realtime callbacks (which fire on the collection key `tableKey("projects", userId)`). Instead, re-use the collection query with `select`:

```typescript
// apps/web/app/(app)/projects/[projectId]/page.tsx (Server Component)
const [project, allProjects] = await Promise.all([
  getProjectById(user.id, projectId),         // for notFound 404 path
  getProjectsForCurrentUser(),                 // hydration source for useQuery initialData
]);
if (!project) notFound();
return <ProjectDetailClient
  userId={user.id}
  projectId={projectId}
  initialProjects={allProjects}
  initialTasks={tasksForProject}
  initialCaptures={capturesForProject}
/>;

// apps/web/components/projects/ProjectDetailClient.tsx (new Client island)
const { data: project } = useQuery({
  queryKey: tableKey("projects", userId),
  queryFn: getProjectsForCurrentUser,
  initialData: initialProjects,
  select: (rows) => rows.find((r) => r.id === projectId),
});
useTableSubscription("projects", userId);     // any project mutation invalidates ['projects', userId]
                                              //   → select re-runs → header re-renders
```

Reasoning for FULL removal of `revalidatePath` in this plan's Server Actions (resolves D-12 open question):
- Realtime echoes drive cache invalidation across all open clients (including the originating tab once optimism settles).
- `revalidatePath` triggers a Server Component re-render which fights `useQuery({ initialData })` — the page-level initialData is only consumed once anyway, and subsequent reads come from TanStack Query cache.
- Keeping `revalidatePath` as "belt-and-suspenders" causes a duplicate refetch every mutation, which is wasted work.
- Removal is safe because Realtime is now the source of cache invalidation truth (RT-04).

    ```typescript
    /**
     * Canonical optimistic reducer for Phase 3 Realtime + useOptimistic.
     *
     * RT-05 dedupe contract: "insert" with an id already in state is a no-op.
     * This is what makes the optimistic + Realtime echo round-trip flicker-free:
     *   1. UI calls addOptimistic({ type: "insert", row: { id: callerUuid, ... } })
     *   2. Server Action persists with that UUID and returns success
     *   3. Realtime echo (INSERT event) arrives → invalidateQueries → refetch
     *   4. Refetched data contains the row with the same id; optimistic state
     *      already has it, so the redux-style merge is a no-op once the
     *      canonical data arrives.
     */
    export type OptimisticAction<T extends { id: string }> =
      | { type: "insert"; row: T }
      | { type: "update"; id: string; patch: Partial<T> }
      | { type: "delete"; id: string }
      | { type: "reorder"; ids: string[] };

    export function optimisticReducer<T extends { id: string }>(
      state: readonly T[],
      action: OptimisticAction<T>,
    ): T[] {
      switch (action.type) {
        case "insert":
          if (state.some((r) => r.id === action.row.id)) return state.slice();
          return [action.row, ...state];
        case "update":
          return state.map((r) =>
            r.id === action.id ? ({ ...r, ...action.patch } as T) : r,
          );
        case "delete":
          return state.filter((r) => r.id !== action.id);
        case "reorder": {
          const map = new Map(state.map((r) => [r.id, r]));
          const out: T[] = [];
          for (const id of action.ids) {
            const row = map.get(id);
            if (row) out.push(row);
          }
          // append any rows not referenced in ids to avoid data loss
          for (const r of state) if (!action.ids.includes(r.id)) out.push(r);
          return out;
        }
        default:
          return state.slice();
      }
    }
    ```
